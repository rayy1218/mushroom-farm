# Run Page Implementation Plan


**Goal:** A run of blocks seen from every module: the Runs tab (`#/business/runs`, the fifth Business tab) listing each run's yield and money, and a run's page (`#/business/runs/<id>`) leading with its profit so far, then its blocks, harvests, sales and costs.

**Architecture:** Approach A of the spec (`docs/specs/2026-11-05-runs-design.md`): nothing is stored. A server module (`server/runs.js`) composes each run on every read from the modules' own functions: Harvest's room view, Batches' list, Sales' summary and its rows, and Costs' run figures through one new marked helper (`runViews`). `server/db.js` gains one marked read (`allRuns`). `server/runs-api.js` serves `GET /api/runs?crop=` and `GET /api/runs/:id`. The pages (`web/runs-page.js`) keep their words in a pure module (`web/runs-math.js`) and format every figure with the owning module's formatter (the One Format Rule). Small marked edits link the run page from Harvest's tab, Costs' runs and Batches' room heads.

**Tech Stack:** Node 22.13+, `node:http`, `node:sqlite`, plain HTML/CSS/JS modules, no build, no npm dependencies, `node --test`.

// The Run page (spec 2026-11-05-runs-design.md): a run of blocks seen from every module, composed on each read from the
// modules' own functions: Harvest's room view (kg, kg per block, BE, flushes, harvests), Batches' list (the run's batches and
// their block cost), Sales' summary and the run's sale lines, Costs' run figures (income, costs, profit, by kind). No table,
// no state, no HTTP.

import { CROP_KEYS } from '../web/crops.js';

export const LATEST_HARVESTS = 8; // the run page's latest picks (the Harvest tab's own count)
const ALL = [-8.64e15, 8.64e15]; // every sale, as Sales' summary reads them

// A refused request: status 400 or 404, field the input it is about (the routes reply { error, field }).
export class RunError extends Error {
  constructor(status, message, field = undefined) {
    super(message);
    this.name = 'RunError';
    this.status = status;
    this.field = field;
  }
}

// The crop filter: null or '' for every run, else one of the crop keys.
function cropOf(crop) {
  if (crop === null || crop === undefined || crop === '') return null;
  if (!CROP_KEYS.includes(crop)) throw new RunError(400, 'Pick a crop', 'crop');
  return crop;
}

// db: the farm database; harvests, batches, sales, costs: the modules' services (createHarvests, createBatches, createSales,
// createCosts), the same ones the app's routes use where a module keeps state (Harvest's and Costs' usage warms).
export function createRuns({ db, harvests, batches, sales, costs }) {
  // Harvest's view of each run of these rooms: Map run id → the room view's run (current or past).
  function harvestViews(roomIds) {
    const out = new Map();
    for (const id of new Set(roomIds)) {
      const v = harvests.roomView(id);
      if (!v) continue; // a room gone between the reads
      for (const run of [v.current, ...v.past]) if (run) out.set(run.id, run);
    }
    return out;
  }

  // A run's row: the list's figures. run: db.allRuns()'s; h: Harvest's view of it; c: Costs' (either may be missing for a
  // run whose room is gone: nothing picked, no money).
  const rowOf = (run, h, c) => ({
    id: run.id, roomId: run.roomId, roomName: run.roomName, crops: run.crops, partial: run.partial,
    startedAt: run.startedAt, endedAt: run.endedAt,
    kg: h?.kg ?? 0, perBlock: h?.perBlock ?? null,
    income: c?.income ?? 0, costs: c ? c.costs : null, profit: c ? c.profit : null,
    pending: c?.pending ?? false, incomplete: c?.incomplete ?? false,
  });

  return {
    // GET /api/runs?crop=: { runs: [row] }, the open runs first, then the ended ones, each newest started first.
    list({ crop = null } = {}) {
      const only = cropOf(crop);
      const all = db.allRuns().filter((r) => only === null || r.crops.some((c) => c.crop === only));
      const h = harvestViews(all.filter((r) => r.roomName !== null).map((r) => r.roomId));
      const c = new Map((all.length ? costs.runViews() : []).map((v) => [v.id, v]));
      const rows = all.map((r) => rowOf(r, h.get(r.id), c.get(r.id)));
      return { runs: [...rows.filter((r) => r.endedAt === null), ...rows.filter((r) => r.endedAt !== null)] };
    },

    // GET /api/runs/:id: the row and the run page's parts; 404 for no such run.
    view(id) {
      const found = db.getRunById(id);
      if (!found) throw new RunError(404, 'Run not found');
      const run = { ...found, roomName: db.getRoom(found.roomId)?.name ?? null };
      const h = run.roomName === null ? null : harvestViews([run.roomId]).get(id) ?? null;
      const [c = null] = costs.runViews({ run: id });
      const { batches: list, blockCost } = batches.list({ run: id });
      const { kg, total, byCrop } = sales.summary({ run: id });
      const lines = db.salesBetween(...ALL).filter((s) => s.runId === id).toReversed().map((s) => ({
        id: s.id, deliveryId: s.deliveryId, soldAt: s.soldAt, buyer: s.buyer, invoiceNo: s.invoiceNo, crop: s.crop, kg: s.kg,
        priceKg: s.priceKg, total: s.total, paidAt: s.paidAt,
      }));
      return {
        ...rowOf(run, h, c),
        profitPerKg: c?.profitPerKg ?? null, kgSold: c?.kgSold ?? 0,
        be: h?.be ?? null, byCrop: h?.byCrop ?? [], flushes: h?.flushes ?? [], missing: h?.missing ?? { dryKg: [], price: [] },
        harvests: (h?.harvests ?? []).slice(0, LATEST_HARVESTS), harvestCount: h?.harvests.length ?? 0,
        batches: list, blockCost,
        sales: { kg, total, byCrop, lines },
        costsByKind: c ? { parts: c.parts, byCategory: c.byCategory, notes: c.notes } : null,
      };
    },
  };
}

// The Run page's routes (spec 2026-11-05-runs-design.md 5), called from api.js: GET /api/runs?crop= and GET /api/runs/:id.
// server/runs.js composes the run from the modules; this reads the query and turns its errors into the API's. Read only:
// it publishes nothing of its own ('runs' is not an event); its Costs service's usage warm, when done, publishes 'costs' so
// a page that waited for the energy asks again.

import { createRuns } from './runs.js';
import { createSales } from './sales.js';
import { createCosts } from './costs.js';
import { localZone } from './usage.js';

const ONE = /^\/api\/runs\/([1-9][0-9]{0,14})$/;

// db: the farm database; events: the bus (null in some tests); harvests and batches: api.js's own services (Harvest's figures
// take Batches' dry kg, and its usage warm publishes 'harvest' there). → async (req, res, path, { send, HttpError, url })
// answering true when the path is one of its routes; its close() stops the Costs service's warms.
export function createRunRoutes({ db, events = null, harvests, batches, now = Date.now, zone = localZone }) {
  const costs = createCosts({ db, now, zone });
  const stopTelling = costs.onChange(() => events?.publish('costs', {}));
  const runs = createRuns({ db, harvests, batches, sales: createSales({ db, now }), costs });

  async function route(req, res, path, { send, HttpError, url }) {
    const one = ONE.exec(path);
    if (path !== '/api/runs' && !one) return false;
    if (req.method !== 'GET') throw new HttpError(404, 'Not found');
    send(res, 200, one ? runs.view(Number(one[1])) : runs.list({ crop: url.searchParams.get('crop') }));
    return true;
  }

  // A 4xx from server/runs.js (RunError) or a module it reads becomes the API's HttpError.
  async function runRoutes(req, res, path, helpers) {
    try {
      return await route(req, res, path, helpers);
    } catch (err) {
      const { HttpError } = helpers;
      if (!(err instanceof HttpError) && Number.isInteger(err?.status) && err.status >= 400 && err.status < 500) {
        throw new HttpError(err.status, err.message, err.field);
      }
      throw err;
    }
  }
  runRoutes.close = () => { stopTelling(); costs.stop(); };
  return runRoutes;
}

  const dates = runDates(run, today);
  const profit = moneyText(run.profit);
  return {
    href: runHash(run.id), label: `${room}, ${dates}`, room, roomHref, crops: cropsText(run), crop: mainCrop(run.crops), dates,

// Reads a path now, again on the events, the stream's return and a timer; paint(reply) gets each new answer, fail(err) a
// failure (the last answer stays on show). → { load, stop } (stop: no more reads, also from inside fail).
function reader(path, { paint, fail, alive }) {
  let seq = 0;
  let timer = null;
  let outdated = false;
  let stopped = false;
  async function load() {
    if (!alive() || outdated || stopped) return;
    clearTimeout(timer);
    const mine = ++seq;
    try {
      const reply = await request('GET', path());
      if (!alive() || mine !== seq) return;
      paint(reply);
    } catch (err) {
      if (!alive() || mine !== seq) return;
      outdated = routeMissing(err);
      fail(err);
      if (outdated || stopped) return;
    }
    clearTimeout(timer);
    timer = setTimeout(load, fallbackMs(REFRESH_MS, live.connected()));
  }
  const reload = createThrottle(() => { if (alive()) load(); });
  const offLive = [
    ...LISTEN.map((type) => live.on(type, () => reload.call())),
    live.on('room', (d) => { if (d.part === 'crop' || d.part === 'room') reload.call(); }), // a run starts or ends; a room's name
    live.onStatus((up) => { if (up) reload.call(); }),
  ];
  return {
    load,
    stop() {
      stopped = true;
      reload.cancel();
      clearTimeout(timer);
      for (const off of offLive) off();
    },
  };
}

  q('[data-retry]').addEventListener('click', () => read.load());
  read.load();

  return function unmount() {
    mountSeq++;
    read.stop();
  };
}
