// A sale is one crop line of a delivery to a buyer; the lines of a delivery share delivery_id (its first line's id) and are
// written together. buyer: the buyer's name at the sale (a rename rewrites it). room_id and run_id null: mixed, not tied.
const SALES = `
CREATE TABLE IF NOT EXISTS buyers (
  id         INTEGER PRIMARY KEY,
  name       TEXT    NOT NULL UNIQUE COLLATE NOCASE,
  phone      TEXT,
  note       TEXT,
  prices     TEXT    NOT NULL DEFAULT '{}',
  archived   INTEGER NOT NULL DEFAULT 0 CHECK (archived IN (0, 1)),
  created_at INTEGER NOT NULL
) STRICT;
CREATE TABLE IF NOT EXISTS sales (
  id          INTEGER PRIMARY KEY,
  delivery_id INTEGER NOT NULL,
  sold_at     INTEGER NOT NULL,
  crop        TEXT    NOT NULL,
  kg          REAL    NOT NULL CHECK (kg > 0 AND kg <= 10000),
  price_kg    REAL    NOT NULL CHECK (price_kg >= 0 AND price_kg <= 1000),
  total       REAL    NOT NULL CHECK (total >= 0),
  buyer       TEXT    NOT NULL,
  buyer_id    INTEGER NOT NULL,
  invoice_no  TEXT,
  room_id     INTEGER,
  run_id      INTEGER,
  note        TEXT,
  paid_at     INTEGER,
  created_at  INTEGER NOT NULL
) STRICT;
CREATE INDEX IF NOT EXISTS sales_by_time ON sales (sold_at);
CREATE INDEX IF NOT EXISTS sales_by_delivery ON sales (delivery_id);
CREATE INDEX IF NOT EXISTS sales_by_buyer ON sales (buyer_id);
CREATE INDEX IF NOT EXISTS sales_by_run ON sales (run_id) WHERE run_id IS NOT NULL;
CREATE INDEX IF NOT EXISTS sales_unpaid ON sales (sold_at) WHERE paid_at IS NULL;`;

// The Sales statements of a database, prepared on first use (after migrate) and kept per connection.
const salesStatements = new WeakMap();
function salesQ(db) {
  let s = salesStatements.get(db);
  if (s) return s;
  s = {
    runAt: db.prepare(`SELECT id FROM runs WHERE room_id = ? AND started_at <= ? AND (ended_at IS NULL OR ended_at > ?)
                       ORDER BY started_at DESC, id DESC LIMIT 1`),
    insertSale: db.prepare(`INSERT INTO sales (delivery_id, sold_at, crop, kg, price_kg, total, buyer, buyer_id, invoice_no, room_id, run_id, note, paid_at, created_at)
                            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`),
    setDeliveryId: db.prepare('UPDATE sales SET delivery_id = ? WHERE id = ?'),
    deliveryLines: db.prepare('SELECT * FROM sales WHERE delivery_id = ? ORDER BY id'),
    deleteDelivery: db.prepare('DELETE FROM sales WHERE delivery_id = ?'),
    setPaid: db.prepare('UPDATE sales SET paid_at = ? WHERE delivery_id = ?'),
    between: db.prepare('SELECT * FROM sales WHERE sold_at BETWEEN ? AND ? ORDER BY sold_at, id'),
    unpaid: db.prepare('SELECT * FROM sales WHERE paid_at IS NULL ORDER BY sold_at, id'),
    clearRoom: db.prepare('UPDATE sales SET room_id = NULL, run_id = NULL WHERE room_id = ?'),
    buyers: db.prepare('SELECT * FROM buyers ORDER BY name COLLATE NOCASE, id'),
    buyer: db.prepare('SELECT * FROM buyers WHERE id = ?'),
    insertBuyer: db.prepare('INSERT INTO buyers (name, phone, note, prices, archived, created_at) VALUES (?, ?, ?, ?, 0, ?)'),
    updateBuyer: db.prepare('UPDATE buyers SET name = ?, phone = ?, note = ?, prices = ?, archived = ? WHERE id = ?'),
    renameSales: db.prepare('UPDATE sales SET buyer = ? WHERE buyer_id = ?'),
    deleteBuyer: db.prepare('DELETE FROM buyers WHERE id = ?'),
    buyerSales: db.prepare('SELECT COUNT(DISTINCT delivery_id) AS n FROM sales WHERE buyer_id = ?'),
  };
  salesStatements.set(db, s);
  return s;
}

const saleOf = (r) => ({
  id: r.id, deliveryId: r.delivery_id, soldAt: r.sold_at, crop: r.crop, kg: r.kg, priceKg: r.price_kg, total: r.total,
  buyer: r.buyer, buyerId: r.buyer_id, invoiceNo: r.invoice_no, roomId: r.room_id, runId: r.run_id, note: r.note,
  paidAt: r.paid_at, createdAt: r.created_at,
});

const buyerOf = (r) => {
  let prices = {};
  try { prices = JSON.parse(r.prices) ?? {}; } catch { /* a bad value reads as none */ }
  return { id: r.id, name: r.name, phone: r.phone, note: r.note, prices, archived: r.archived === 1, createdAt: r.created_at };
};

    //      buyer, buyerId, invoiceNo, roomId, runId, note, paidAt, createdAt }; a buyer { id, name, phone, note, prices,
    //      archived, createdAt }. The store (server/sales.js) checks every value first. ----

    // The id of the room's run open at ms (started at or before it, not ended by it), or null.
    runAt(roomId, ms) {
      return salesQ(db).runAt.get(roomId, ms, ms)?.id ?? null;
    },

    // Writes a delivery: { id (null: new), soldAt, buyerId, buyer, invoiceNo, note, lines: [{ crop, kg, priceKg, total,
    // roomId, runId }] } at `at`. An edit replaces the lines and keeps the id and the paid time. → the delivery's id, or null
    // when id names no delivery.
    saveDelivery({ id = null, soldAt, buyerId, buyer, invoiceNo, note, lines }, at) {
      const s = salesQ(db);
      let deliveryId = id;
      inTransaction(() => {
        let paidAt = null;
        if (id !== null) {
          const old = s.deliveryLines.all(id);
          if (!old.length) {
            deliveryId = null;
            return;
          }
          paidAt = old[0].paid_at;
          s.deleteDelivery.run(id);
        }
        for (const l of lines) {
          const rowId = Number(s.insertSale.run(deliveryId ?? 0, soldAt, l.crop, l.kg, l.priceKg, l.total, buyer, buyerId,
            invoiceNo, l.roomId, l.runId, note, paidAt, at).lastInsertRowid);
          if (deliveryId === null) {
            deliveryId = rowId;
            s.setDeliveryId.run(rowId, rowId);
          }
        }
      });
      return deliveryId;
    },

    // A delivery's lines in the order written ([] for none).
    deliveryLines(id) {
      return salesQ(db).deliveryLines.all(id).map(saleOf);
    },

    // → true when a delivery went.
    deleteDelivery(id) {
      return salesQ(db).deleteDelivery.run(id).changes > 0;
    },

    // Paid at ms, or unpaid (null), on every line. → true when the delivery exists.
    setDeliveryPaid(id, paidAt) {
      return salesQ(db).setPaid.run(paidAt, id).changes > 0;
    },

    // The sale lines sold in [from, to] (ms), oldest first.
    salesBetween(from, to) {
      return salesQ(db).between.all(from, to).map(saleOf);
    },

    // Every unpaid sale line, oldest first.
    unpaidSales() {
      return salesQ(db).unpaid.all().map(saleOf);
    },

    // The buyers A–Z (without regard to case).
    listBuyers() {
      return salesQ(db).buyers.all().map(buyerOf);
    },

    // A buyer, or null.
    getBuyer(id) {
      const r = salesQ(db).buyer.get(id);
      return r ? buyerOf(r) : null;
    },

    // A new buyer: { name, phone, note, prices }. Throws on a name taken (UNIQUE, without regard to case). → the buyer.
    insertBuyer({ name, phone, note, prices }, at) {
      const id = Number(salesQ(db).insertBuyer.run(name, phone, note, JSON.stringify(prices ?? {}), at).lastInsertRowid);
      return this.getBuyer(id);
    },

    // Changes the given fields of a buyer; a new name is written on their sales too, in one transaction. → the buyer, or null.
    updateBuyer(id, fields) {
      const s = salesQ(db);
      const old = this.getBuyer(id);
      if (!old) return null;
      const b = { ...old, ...fields };
      inTransaction(() => {
        s.updateBuyer.run(b.name, b.phone, b.note, JSON.stringify(b.prices ?? {}), b.archived ? 1 : 0, id);
        if (b.name !== old.name) s.renameSales.run(b.name, id);
      });
      return this.getBuyer(id);
    },

    deleteBuyer(id) {
      return salesQ(db).deleteBuyer.run(id).changes > 0;
    },

    buyerSaleCount(id) {
      return salesQ(db).buyerSales.get(id).n;

// Sales' pure parts (spec 2026-11-05-sales-design.md 2, 4 and 5): the checks of a delivery, a buyer and a paid body (each
// throws SalesError with the field of the box), deliveries grouped from their lines, what is owed, the figures per crop and
// per buyer, the summary Costs reads, and the CSV rows. No database, no HTTP.

import { CROP_KEYS, cropShortName } from '../web/crops.js';
import { localDate, localTime } from './task-rules.js';
import { round } from './harvest-math.js';
import { cell } from './csv.js';

export const MAX_KG = 10_000;
export const MAX_PRICE = 1_000;
export const MAX_LINES = 9;
const FIRST_MS = Date.UTC(2020, 0, 1) - 8 * 3_600_000; // 1 Jan 2020 at the farm
const GRACE_MS = 60_000; // a minute for clocks

// A refused request: status 400, 404 or 409; field the input it is about.
export class SalesError extends Error {
  constructor(status, message, field = undefined) {
    super(message);
    this.name = 'SalesError';
    this.status = status;
    this.field = field;
  }
}

const isObject = (v) => v !== null && typeof v === 'object' && !Array.isArray(v);
const isNumber = (v) => typeof v === 'number' && Number.isFinite(v);
const isId = (v) => Number.isSafeInteger(v) && v > 0;
const cropOrder = (a, b) => CROP_KEYS.indexOf(a.crop) - CROP_KEYS.indexOf(b.crop);
const sum = (list, key) => round(list.reduce((s, x) => s + x[key], 0), 2);

// Optional text: empty or missing is null; otherwise trimmed and at most max characters.
function textOf(v, max, field, words) {
  if (v === undefined || v === null) return null;
  if (typeof v !== 'string') throw new SalesError(400, words, field);
  const t = v.trim();
  if (t.length > max) throw new SalesError(400, words, field);
  return t || null;
}

// A time: a whole number of ms from 1 Jan 2020 at the farm to now (a minute's grace).
function timeOf(v, now, field, future) {
  if (!Number.isSafeInteger(v) || v < FIRST_MS) throw new SalesError(400, 'Give the date', field);
  if (v > now + GRACE_MS) throw new SalesError(400, future, field);
  return v;
}

// A delivery's body: { soldAt, buyerId, invoiceNo?, note?, lines: [{ crop, kg, priceKg, roomId? }] }. → { soldAt, buyerId,
// invoiceNo, note, lines: [{ crop, kg, priceKg, total, roomId }] } in the body's order (a field names a line by its index).
// The store checks the buyer and the rooms exist.
export function deliveryBody(body, { now }) {
  if (!isObject(body)) throw new SalesError(400, 'Body must be a JSON object');
  const soldAt = timeOf(body.soldAt, now, 'soldAt', "The sale can't be in the future");
  if (!isId(body.buyerId)) throw new SalesError(400, 'Pick a buyer', 'buyerId');
  const invoiceNo = textOf(body.invoiceNo, 40, 'invoiceNo', 'The reference is at most 40 characters');
  const note = textOf(body.note, 200, 'note', 'A note is at most 200 characters');
  if (!Array.isArray(body.lines) || body.lines.length === 0) throw new SalesError(400, 'Add at least one crop', 'lines');
  if (body.lines.length > MAX_LINES) throw new SalesError(400, `At most ${MAX_LINES} crops in a sale`, 'lines');
  const seen = new Set();
  const lines = body.lines.map((l, i) => {
    const at = `lines.${i}`;
    if (!isObject(l) || !CROP_KEYS.includes(l.crop)) throw new SalesError(400, 'Pick a crop', `${at}.crop`);
    if (seen.has(l.crop)) throw new SalesError(400, 'This crop is on another line', `${at}.crop`);
    seen.add(l.crop);
    const kg = isNumber(l.kg) ? round(l.kg, 2) : NaN;
    if (!(kg > 0 && kg <= MAX_KG)) throw new SalesError(400, 'kg must be a number over 0 up to 10,000', `${at}.kg`);
    const priceKg = isNumber(l.priceKg) ? round(l.priceKg, 2) : NaN;
    if (!(priceKg >= 0 && priceKg <= MAX_PRICE)) throw new SalesError(400, 'RM per kg must be 0 to 1,000', `${at}.priceKg`);
    const roomId = l.roomId === undefined || l.roomId === null ? null : l.roomId;
    if (roomId !== null && !isId(roomId)) throw new SalesError(400, 'Pick a room or Mixed', `${at}.roomId`);
    return { crop: l.crop, kg, priceKg, total: round(kg * priceKg, 2), roomId };
  });
  return { soldAt, buyerId: body.buyerId, invoiceNo, note, lines };
}

// A buyer's body: { name, phone?, note?, prices? } (partial: only the keys given, and archived). prices: { <crop>: RM per kg }
// with an empty value (null, '') left out. → the fields.
export function buyerBody(body, { partial = false } = {}) {
  if (!isObject(body)) throw new SalesError(400, 'Body must be a JSON object');
  const out = {};
  if (!partial || 'name' in body) {
    const name = typeof body.name === 'string' ? body.name.trim() : '';
    if (!name) throw new SalesError(400, 'Give the buyer a name', 'name');
    if (name.length > 60) throw new SalesError(400, 'A name is at most 60 characters', 'name');
    out.name = name;
  }
  if (!partial || 'phone' in body) out.phone = textOf(body.phone, 30, 'phone', 'A phone number is at most 30 characters');
  if (!partial || 'note' in body) out.note = textOf(body.note, 200, 'note', 'A note is at most 200 characters');
  if (!partial || 'prices' in body) {
    const prices = body.prices ?? {};
    if (!isObject(prices)) throw new SalesError(400, 'Prices must be per crop', 'prices');
    out.prices = {};
    for (const [crop, v] of Object.entries(prices)) {
      if (!CROP_KEYS.includes(crop)) throw new SalesError(400, 'Unknown crop', 'prices');
      if (v === null || v === '') continue;
      const p = isNumber(v) ? round(v, 2) : NaN;
      if (!(p >= 0 && p <= MAX_PRICE)) throw new SalesError(400, 'RM per kg must be 0 to 1,000', `prices.${crop}`);
      out.prices[crop] = p;
    }
  }
  if (partial && 'archived' in body) {
    if (typeof body.archived !== 'boolean') throw new SalesError(400, 'archived must be true or false', 'archived');
    out.archived = body.archived;
  }
  return out;
}

// Mark paid or unpaid: { paidAt: ms | null }. Paid not before the sale's farm day and not in the future. → ms or null.
export function paidBody(body, { soldAt, now }) {
  if (!isObject(body) || !('paidAt' in body)) throw new SalesError(400, 'Give the day it was paid', 'paidAt');
  if (body.paidAt === null) return null;
  const paidAt = timeOf(body.paidAt, now, 'paidAt', "Paid can't be in the future");
  if (localDate(paidAt) < localDate(soldAt)) throw new SalesError(400, 'Paid before the sale', 'paidAt');
  return paidAt;
}

// Deliveries from their lines (db rows), newest first. rooms: Map id → { name }; runs: Map id → { startedAt }; buyers: Map
// id → { archived }. A delivery: { id, soldAt, buyerId, buyer, buyerArchived, invoiceNo, note, paidAt, kg, total, lines }.
export function deliveriesOf(rows, { rooms = new Map(), runs = new Map(), buyers = new Map() } = {}) {
  const groups = new Map();
  for (const r of rows) groups.set(r.deliveryId, [...(groups.get(r.deliveryId) ?? []), r]);
  return [...groups.entries()].map(([id, own]) => {
    const [h] = own;
    const lines = [...own].sort(cropOrder).map((l) => ({
      id: l.id, crop: l.crop, kg: l.kg, priceKg: l.priceKg, total: l.total, roomId: l.roomId,
      roomName: l.roomId === null ? null : rooms.get(l.roomId)?.name ?? null, runId: l.runId,
      runStartedAt: l.runId === null ? null : runs.get(l.runId)?.startedAt ?? null,
    }));
    return {
      id, soldAt: h.soldAt, buyerId: h.buyerId, buyer: h.buyer, buyerArchived: buyers.get(h.buyerId)?.archived ?? false,
      invoiceNo: h.invoiceNo, note: h.note, paidAt: h.paidAt, kg: sum(own, 'kg'), total: sum(own, 'total'), lines,
    };
  }).sort((a, b) => b.soldAt - a.soldAt || b.id - a.id);
}

// What is owed: the unpaid deliveries (any order) by buyer, most owed first, each buyer's deliveries oldest first. → { total,
// count, oldestAt (null when none), buyers: [{ buyerId, buyer, total, count, oldestAt, deliveries }] }.
export function owedOf(unpaid) {
  const oldestFirst = [...unpaid].sort((a, b) => a.soldAt - b.soldAt || a.id - b.id);
  const by = new Map();
  for (const d of oldestFirst) by.set(d.buyerId, [...(by.get(d.buyerId) ?? []), d]);
  const buyers = [...by.entries()].map(([buyerId, list]) => ({
    buyerId, buyer: list[list.length - 1].buyer, total: sum(list, 'total'), count: list.length, oldestAt: list[0].soldAt, deliveries: list,
  })).sort((a, b) => b.total - a.total || a.buyer.localeCompare(b.buyer));
  return { total: sum(oldestFirst, 'total'), count: oldestFirst.length, oldestAt: oldestFirst[0]?.soldAt ?? null, buyers };
}

// Per crop over a period: the sale lines and the harvests in it, the Settings → Yield prices. → [{ crop, kg, harvestedKg,
// total, avgPriceKg (null with no kg sold), expectedPriceKg (null when not set) }] in the crop list's order.
export function byCropOf(rows, harvests, cropYield) {
  const crops = CROP_KEYS.filter((c) => rows.some((r) => r.crop === c) || harvests.some((h) => h.crop === c && h.kg > 0));
  return crops.map((crop) => {
    const own = rows.filter((r) => r.crop === crop);
    const kg = sum(own, 'kg');
    const total = sum(own, 'total');
    const expected = cropYield?.[crop]?.priceKg;
    return {
      crop, kg, harvestedKg: sum(harvests.filter((h) => h.crop === crop), 'kg'), total,
      avgPriceKg: kg > 0 ? round(total / kg, 2) : null, expectedPriceKg: isNumber(expected) ? expected : null,
    };
  });
}

// Per buyer over a period's deliveries, with what each owes all time (owedOf). Most income first.
export function byBuyerOf(deliveries, owed) {
  const by = new Map();
  for (const d of deliveries) by.set(d.buyerId, [...(by.get(d.buyerId) ?? []), d]);
  return [...by.entries()].map(([buyerId, list]) => ({
    buyerId, buyer: list[0].buyer, kg: sum(list, 'kg'), total: sum(list, 'total'),
    owed: owed.buyers.find((b) => b.buyerId === buyerId)?.total ?? 0,
  })).sort((a, b) => b.total - a.total || a.buyer.localeCompare(b.buyer));
}

// The summary Costs reads (contract section 5): exactly { kg, total, byCrop: [{ crop, kg, total }], byRun: [{ runId, kg,
// total }] }; a line with no run is in the totals and byCrop only.
export function summaryOf(rows) {
  const group = (list, keyOf) => {
    const by = new Map();
    for (const r of list) by.set(keyOf(r), [...(by.get(keyOf(r)) ?? []), r]);
    return by;
  };
  const byCrop = [...group(rows, (r) => r.crop)].map(([crop, own]) => ({ crop, kg: sum(own, 'kg'), total: sum(own, 'total') })).sort(cropOrder);
  const byRun = [...group(rows.filter((r) => r.runId !== null), (r) => r.runId)].map(([runId, own]) => ({ runId, kg: sum(own, 'kg'), total: sum(own, 'total') }))
    .sort((a, b) => a.runId - b.runId);
  return { kg: sum(rows, 'kg'), total: sum(rows, 'total'), byCrop, byRun };
}

export const CSV_HEADER = 'date,time,buyer,reference,crop,room,run started,kg,price per kg,total,paid on,note';

// The CSV's lines (the header first): one row per sale line, oldest first.
export function csvRows(deliveries) {
  const rows = [...deliveries].sort((a, b) => a.soldAt - b.soldAt || a.id - b.id).flatMap((d) => d.lines.map((l) => [
    localDate(d.soldAt), localTime(d.soldAt), d.buyer, d.invoiceNo, cropShortName(l.crop), l.roomName ?? 'Mixed',
    l.runStartedAt === null ? '' : localDate(l.runStartedAt), l.kg, l.priceKg, l.total, d.paidAt === null ? '' : localDate(d.paidAt), d.note,
  ].map(cell).join(',')));
  return [CSV_HEADER, ...rows];
}

// Sales (spec 2026-11-05-sales-design.md 2 to 5): deliveries recorded, edited, deleted and marked paid; buyers; the page's
// view, the myGAP sheet's range, the summary Costs reads and the CSV. No HTTP.

import { readCropYield } from './settings.js';
import { localDate, dayStart, addDays, daysBetween } from './task-rules.js';
import { buckets, round } from './harvest-math.js';
import {
  SalesError, deliveryBody, buyerBody, paidBody, deliveriesOf, owedOf, byCropOf, byBuyerOf, summaryOf, csvRows,
} from './sales-math.js';

export const RANGES = [7, 30, 90, 365];
const MAX_RANGE_DAYS = 400;
const DATE = /^\d{4}-\d{2}-\d{2}$/;
const ALL = [-8.64e15, 8.64e15];

// A farm date (YYYY-MM-DD) that is a real day, else a 400 on field.
function dateOf(v, field) {
  const ms = typeof v === 'string' && DATE.test(v) ? dayStart(v) : NaN; // '2026-13-01' parses as NaN
  if (!Number.isFinite(ms) || localDate(ms) !== v) throw new SalesError(400, 'Give a date as YYYY-MM-DD', field);
  return v;
}

// db: the farm database; now injectable. log: a listener that throws.
export function createSales({ db, now = Date.now, log = console.error }) {
  const listeners = new Set();

  function tell() {
    for (const fn of [...listeners]) {
      try {
        fn();
      } catch (err) {
        log(err);
      }
    }
  }

  // The rows' deliveries with their rooms', runs' and buyers' words.
  function deliveries(rows) {
    const rooms = new Map(db.listRooms().map((r) => [r.id, r]));
    const runs = new Map();
    for (const r of rows) if (r.runId !== null && !runs.has(r.runId)) runs.set(r.runId, db.getRunById(r.runId));
    const buyers = new Map(db.listBuyers().map((b) => [b.id, b]));
    return deliveriesOf(rows, { rooms, runs, buyers });
  }

  function delivery(id) {
    const rows = db.deliveryLines(id);
    return rows.length ? deliveries(rows)[0] : null;
  }

  function existing(id) {
    const rows = db.deliveryLines(id);
    if (!rows.length) throw new SalesError(404, 'Sale not found');
    return rows[0];
  }

  // A body checked against the database: the buyer exists and is active (or is the sale's own: keep), the rooms exist;
  // each line's run is the room's run open at the sale.
  function checked(body, { keep = null } = {}) {
    const d = deliveryBody(body, { now: now() });
    const buyer = db.getBuyer(d.buyerId);
    if (!buyer) throw new SalesError(400, 'Pick a buyer', 'buyerId');
    if (buyer.archived && buyer.id !== keep) throw new SalesError(400, 'This buyer is archived', 'buyerId');
    const lines = d.lines.map((l, i) => {
      if (l.roomId !== null && !db.getRoom(l.roomId)) throw new SalesError(400, 'That room no longer exists', `lines.${i}.roomId`);
      return { ...l, runId: l.roomId === null ? null : db.runAt(l.roomId, d.soldAt) };
    });
    return { ...d, buyer: buyer.name, lines };
  }

  // A name not used by another buyer (without regard to case), else 409.
  function nameFree(name, id = null) {
    const taken = db.listBuyers().some((b) => b.id !== id && b.name.toLowerCase() === name.toLowerCase());
    if (taken) throw new SalesError(409, 'A buyer with this name exists', 'name');
  }

  // The farm days of a period: one of RANGES, ending today (the Harvest page's ranges).
  function period(days) {
    if (!RANGES.includes(days)) throw new SalesError(400, 'days must be 7, 30, 90 or 365', 'days');
    const { from, to } = buckets([], { days, to: now() });
    return { from, to };
  }

  const rowsBetween = (from, to) => db.salesBetween(dayStart(from), dayStart(addDays(to, 1)) - 1);

  function buyerView(b, owed) {
    return { ...b, sales: db.buyerSaleCount(b.id), owed: owed.buyers.find((o) => o.buyerId === b.id)?.total ?? 0 };
  }

  return {
    // Record a sale: { soldAt, buyerId, invoiceNo?, note?, lines }. → the delivery.
    record(body) {
      const id = db.saveDelivery(checked(body), now());
      tell();
      return delivery(id);
    },

    // Replace a delivery's fields and lines; its paid time stays (a new date after the paid day is refused). → the delivery.
    edit(id, body) {
      const old = existing(id);
      const d = checked(body, { keep: old.buyerId });
      if (old.paidAt !== null && localDate(d.soldAt) > localDate(old.paidAt)) throw new SalesError(400, 'Paid before this date', 'soldAt');
      db.saveDelivery({ ...d, id }, now());
      tell();
      return delivery(id);
    },

    remove(id) {
      existing(id);
      db.deleteDelivery(id);
      tell();
    },

    // { paidAt: ms | null }. → the delivery.
    setPaid(id, body) {
      const old = existing(id);
      db.setDeliveryPaid(id, paidBody(body, { soldAt: old.soldAt, now: now() }));
      tell();
      return delivery(id);
    },

    // The Sales page (spec 5): { from, to, days, deliveries, totals: { kg, total, harvestedKg, count }, owed, byCrop, byBuyer,
    // buyerCount }.
    view({ days = 30 } = {}) {
      const { from, to } = period(days);
      const rows = rowsBetween(from, to);
      const list = deliveries(rows);
      const harvests = db.harvestsBetween(dayStart(from), dayStart(addDays(to, 1)) - 1);
      const owed = owedOf(deliveries(db.unpaidSales()));
      return {
        from, to, days, deliveries: list,
        totals: {
          kg: round(rows.reduce((s, r) => s + r.kg, 0), 2), total: round(rows.reduce((s, r) => s + r.total, 0), 2),
          harvestedKg: round(harvests.reduce((s, h) => s + h.kg, 0), 2), count: list.length,
        },
        owed, byCrop: byCropOf(rows, harvests, readCropYield(db)), byBuyer: byBuyerOf(list, owed),
        buyerCount: db.listBuyers().filter((b) => !b.archived).length, // none: the page says to add the first
      };
    },

    // The myGAP sheet's sale lines from one farm day to another (at most 400 days), oldest first.
    range({ from, to }) {
      dateOf(from, 'from');
      dateOf(to, 'to');
      if (from > to) throw new SalesError(400, 'from must be on or before to', 'from');
      if (daysBetween(from, to) + 1 > MAX_RANGE_DAYS) throw new SalesError(400, 'At most 400 days', 'from');
      const lines = [...deliveries(rowsBetween(from, to))].reverse().flatMap((d) => d.lines.map((l) => ({
        date: localDate(d.soldAt), soldAt: d.soldAt, crop: l.crop, roomName: l.roomName, runStartedAt: l.runStartedAt,
        kg: l.kg, priceKg: l.priceKg, total: l.total, invoiceNo: d.invoiceNo, buyer: d.buyer,
      })));
      return { from, to, lines };
    },

    // Costs' summary (contract 5): from and to optional farm days (none: all time), run optional.
    summary({ from = null, to = null, run = null } = {}) {
      const start = from === null ? ALL[0] : dayStart(dateOf(from, 'from'));
      const end = to === null ? ALL[1] : dayStart(addDays(dateOf(to, 'to'), 1)) - 1;
      const rows = db.salesBetween(start, end).filter((r) => run === null || r.runId === run);
      return summaryOf(rows);
    },

    // The period's sale lines as CSV: { name: 'sales <from>–<to>.csv', text } (the route adds the BOM).
    csv(days = 30) {
      const { from, to } = period(days);
      return { name: `sales ${from}–${to}.csv`, text: `${csvRows(deliveries(rowsBetween(from, to))).join('\r\n')}\r\n` };
    },

    // The buyers A–Z with their deliveries' count and what they owe.
    buyers() {
      const owed = owedOf(deliveries(db.unpaidSales()));
      return db.listBuyers().map((b) => buyerView(b, owed));
    },

    addBuyer(body) {
      const fields = buyerBody(body);
      nameFree(fields.name);
      const b = db.insertBuyer(fields, now());
      tell();
      return buyerView(b, owedOf([]));
    },

    editBuyer(id, body) {
      if (!db.getBuyer(id)) throw new SalesError(404, 'Buyer not found');
      const fields = buyerBody(body, { partial: true });
      if (fields.name !== undefined) nameFree(fields.name, id);
      db.updateBuyer(id, fields);
      tell();
      return this.buyers().find((b) => b.id === id);
    },

    removeBuyer(id) {
      if (!db.getBuyer(id)) throw new SalesError(404, 'Buyer not found');
      if (db.buyerSaleCount(id) > 0) throw new SalesError(409, 'Archive this buyer instead: it has sales');
      db.deleteBuyer(id);
      tell();
    },

    // fn() after any sale or buyer is written; → a function that stops it.
    onChange(fn) {
      listeners.add(fn);
      return () => listeners.delete(fn);
    },
  };
}

// answering true when the path is one of its routes.
export function createSalesRoutes({ db, events = null, now = Date.now }) {
  const sales = createSales({ db, now });
  sales.onChange(() => events?.publish('sales', {}));

  async function route(req, res, path, { send, readJson, HttpError, url }) {
    const notFound = () => new HttpError(404, 'Not found');
    const noContent = () => { res.writeHead(204); res.end(); };

    // days: digits only; the store checks it is one of the ranges.

const NB = ' ';
const DASH = '—';
const DAY_MS = 86_400_000;
const FARM_OFFSET_MS = 8 * 3_600_000;
const WEEKDAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
export const UNLIKELY_TIMES = 3;
export const UNLIKELY_KG = 500;
const finite = (x) => typeof x === 'number' && Number.isFinite(x);
const two = (n) => n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2, useGrouping: false }); // a box's value
const sheet2 = (n) => n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); // the sheet's figures, grouped as Harvest's
const num = (text) => {
  const t = String(text ?? '').trim().replace(',', '.');
  return t === '' || !/^\d*\.?\d+$/.test(t) ? NaN : Number(t);
};

export const priceText = (v) => rmText(v);

// Whole farm days from a sale to now: "today", "1 day", "12 days".
export function ageText(ms, now) {
  const days = Math.round((Date.parse(farmDate(now)) - Date.parse(farmDate(ms))) / DAY_MS);
  return days <= 0 ? 'today' : days === 1 ? '1 day' : `${days} days`;
}

const shortDay = (ms) => {
  const date = farmDate(ms);
  return `${WEEKDAYS[new Date(`${date}T00:00:00Z`).getUTCDay()]} ${dayMonth(date)}`;
};

// "Grey oyster 12.00 kg · Pink oyster 4.50 kg"
export const linesShort = (lines) => lines.map((l) => `${cropShortName(l.crop)} ${kgText(l.kg, { exact: true })}`).join(DOT);

export const paidMark = (d) => (d.paidAt === null || d.paidAt === undefined
  ? { text: 'Unpaid', unpaid: true }
  : { text: `Paid ${dayMonth(farmDate(d.paidAt)).replace(' ', NB)}`, unpaid: false });

export function pageTiles(sales, now) {
  const { totals, owed } = sales;
  const before = owed.oldestAt !== null && ageText(owed.oldestAt, now) !== 'today'
    && Math.round((Date.parse(farmDate(now)) - Date.parse(farmDate(owed.oldestAt))) / DAY_MS) >= sales.days;
  const owedLine = owed.count === 0 ? 'All paid'
    : `${before ? `all time${DOT}` : ''}${owed.count} ${owed.count === 1 ? 'sale' : 'sales'}${DOT}oldest ${ageText(owed.oldestAt, now)}`;
  return [
    { label: 'Sold', value: kgText(totals.kg), line: `of ${kgText(totals.harvestedKg)} harvested` },
    { label: 'Income', value: rmText(totals.total) },
    { label: 'Owed', value: rmText(owed.total), line: owedLine },

}

export const CROP_COLUMNS = ['Crop', 'Sold', 'Harvested', 'Income', 'Average RM per kg', 'Expected RM per kg'];
export const BUYER_COLUMNS = ['Buyer', 'kg', 'Income', 'Owed'];

// The average price against the expected one, in words (no colour verdict); null when either is missing.
function diffText(avg, expected) {
  if (!finite(avg) || !finite(expected)) return null;
  const d = Math.round((avg - expected) * 100) / 100;
  if (Math.abs(d) < 0.005) return 'as expected';
  return `${rmText(Math.abs(d))} ${d < 0 ? 'under' : 'over'} expected`;
}

export const cropRows = (sales) => (sales.byCrop ?? []).map((c) => ({
  crop: c.crop, name: cropShortName(c.crop),
  cells: [kgText(c.kg), kgText(c.harvestedKg), rmText(c.total), rmText(c.avgPriceKg), rmText(c.expectedPriceKg)],
  diff: diffText(c.avgPriceKg, c.expectedPriceKg),
}));

export const buyerRows = (sales) => (sales.byBuyer ?? []).map((b) => ({ name: b.buyer, cells: [kgText(b.kg), rmText(b.total), rmText(b.owed)] }));

export const emptyText = (days) => `No sales in the last ${periodWords(days)}.`;

export function lineTotal(kg, priceKg) {
  const [k, p] = [num(kg), num(priceKg)];
  return Number.isNaN(k) || Number.isNaN(p) ? '' : rmText(Math.round(k * p * 100) / 100);
}

// The dialog's foot: the lines' kg and total ("16.50 kg · RM 214.50"), the lines that read as numbers.
export function footText(lines) {
  const ok = lines.map((l) => [num(l.kg), num(l.priceKg)]).filter(([k, p]) => !Number.isNaN(k) && !Number.isNaN(p));
  const kg = ok.reduce((s, [k]) => s + k, 0);
  const total = ok.reduce((s, [k, p]) => s + Math.round(k * p * 100) / 100, 0);
  return `${kgText(kg, { exact: true })}${DOT}${rmText(total)}`;
}

// RM per kg for a crop's line: the buyer's usual price, else the expected price (marked), else empty.
export function prefill(crop, buyer, cropYield) {
  const own = buyer?.prices?.[crop];
  if (finite(own)) return { value: two(own), expected: false };
  const expected = cropYield?.[crop]?.priceKg;
  if (finite(expected)) return { value: two(expected), expected: true };
  return { value: '', expected: false };
}

// The questions asked once (not refusals): RM per kg over 3 times the crop's expected price, kg over 500.
export function unlikely(lines, cropYield) {
  return lines.flatMap((l, i) => {
    const out = [];
    const expected = cropYield?.[l.crop]?.priceKg;
    if (finite(expected) && expected > 0 && l.priceKg > UNLIKELY_TIMES * expected) {
      const times = (Math.round((l.priceKg / expected) * 10) / 10).toFixed(1);
      out.push({ field: `lines.${i}.priceKg`, message: `That's ${times} times the expected ${rmText(expected)} per kg. Is ${rmText(l.priceKg)} right?` });
    }
    if (l.kg > UNLIKELY_KG) out.push({ field: `lines.${i}.kg`, message: `Is ${kgText(l.kg, { exact: true })} right?` });
    return out;
  });
}

// The POST/PUT body from the dialog's boxes: { date, buyerId, invoiceNo, note, lines: [{ crop, kg, priceKg, roomId }] } (all
// text). Today's date is now; another day is 12:00 at the farm. → the body, or { error, field }.
export function saleBody(v, { now }) {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(v.date ?? '')) return { error: 'Give the date', field: 'date' };
  if (!/^[1-9]\d*$/.test(String(v.buyerId ?? ''))) return { error: 'Pick a buyer', field: 'buyerId' };
  const soldAt = v.date === farmDate(now) ? now : Date.parse(`${v.date}T12:00:00Z`) - FARM_OFFSET_MS;
  const lines = [];
  for (const [i, l] of (v.lines ?? []).entries()) {
    const kg = num(l.kg);
    if (Number.isNaN(kg)) return { error: 'Give the kg as a number', field: `lines.${i}.kg` };
    const priceKg = num(l.priceKg);
    if (Number.isNaN(priceKg)) return { error: 'Give RM per kg as a number', field: `lines.${i}.priceKg` };
    lines.push({ crop: l.crop, kg, priceKg, roomId: l.roomId ? Number(l.roomId) : null });
  }
  const text = (s) => (String(s ?? '').trim() || null);
  return { soldAt, buyerId: Number(v.buyerId), invoiceNo: text(v.invoiceNo), note: text(v.note), lines };
}

export const deleteTitle = (d) => `Delete the sale of ${dayMonth(farmDate(d.soldAt))} to ${d.buyer}?`;
export const deleteLead = (d) => `${rmText(d.total)}${DOT}${d.lines.map((l) => `${cropShortName(l.crop)} ${kgText(l.kg, { exact: true })}`).join(', ')}.`;

// The myGAP 4-6 sheet's rows: Harvest's rows (sheetRows) and the sale lines together by date, a day's harvest rows first
// then its sale lines by time. A sale row: Date, Crop, Block / location (the room and run as Harvest's, or "Mixed"),
// Harvested blank, Sold, Price / kg, Total, Invoice No., Buyer.
export function sheetWithSales(records, lines) {
  const list = Array.isArray(records) ? records : [];
  const harvest = sheetRows(list).map((cells, i) => ({ date: list[i].date, kind: 0, at: i, cells }));
  const sales = (Array.isArray(lines) ? lines : []).map((l, i) => {
    const [base] = sheetRows([{ date: l.date, roomName: l.roomName ?? '', crop: l.crop, runStartedAt: l.runStartedAt, kg: null }]);
    const where = l.roomName ? base[2] : 'Mixed';
    return { date: l.date, kind: 1, at: l.soldAt * 1000 + i, cells: [base[0], base[1], where, '', sheet2(l.kg), sheet2(l.priceKg), sheet2(l.total), l.invoiceNo ?? '', l.buyer] };
  });
  return [...harvest, ...sales].sort((a, b) => a.date.localeCompare(b.date) || a.kind - b.kind || a.at - b.at).map((r) => r.cells);
}

// The sheet's Total row: harvested kg, sold kg and the RM total ('' when the sales could not be read: lines null).
export function sheetTotals(farm, lines) {
  const harvested = finite(farm?.totals?.kg) ? sheet2(farm.totals.kg) : '';
  if (!Array.isArray(lines)) return { harvested, sold: '', total: '' };
  const sold = lines.reduce((s, l) => s + l.kg, 0);
  const total = lines.reduce((s, l) => s + l.total, 0);
  return { harvested, sold: sheet2(Math.round(sold * 100) / 100), total: sheet2(Math.round(total * 100) / 100) };
}

function table(caption, columns, rows, head) {
  const wrap = el('div', 'us-table-wrap');
  const t = el('table', 'us-table sl-table');
  t.append(el('caption', 'sr-only', caption));
  const tr = el('tr');
  columns.forEach((c, i) => {
    const th = el('th', i ? 'num' : '', c);
    th.scope = 'col';
    tr.append(th);
  });
  const thead = el('thead');
  thead.append(tr);
  const body = el('tbody');
  for (const r of rows) {
    const row = el('tr');
    const th = el('th');
    th.scope = 'row';
    const name = el('span', 'sl-name');
    name.append(...head(r));
    th.append(name);
    if (r.diff) th.append(el('span', 'sl-diff', r.diff));
    row.append(th, ...r.cells.map((c) => el('td', 'num', c)));
    body.append(row);
  }
  t.append(thead, body);
  wrap.append(t);
  return wrap;
}

function field({ id, label, kind = 'text', value = '', hint = null, suffix = null, prefix = null, options = null }) {
