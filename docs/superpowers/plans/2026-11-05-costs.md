# Costs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Costs module: costs entered by hand (one-offs and monthly items), read live with Batches' block costs, Inventory's purchases, Sales and the measured energy and water, giving profit per month, per run and per kg on `#/business/costs`.

**Architecture:** Approach A of the spec (`docs/superpowers/specs/2026-11-05-costs-design.md`): only the `costs` rows are stored (database version 24); every figure is worked out on each read by a pure module (`server/costs-math.js`) from the costs, the runs, the harvests, the usage days (`roomDays`, `cachedOnly`, with a background warm as Harvest has) and the other modules' tables, which one file reads by the contract's column names (`server/costs-sources.js`, empty when a table does not exist yet). A service (`server/costs.js`) checks and writes, and builds the month view and the contract's summary; `server/costs-api.js` is its routes. The page is a plain module (`web/costs-page.js`) with its pure parts in `web/costs-math.js` and its dialogs in `web/costs-dialogs.js`.

**Tech Stack:** Node 22.13+, `node:http`, `node:sqlite`, plain HTML/CSS/JS modules, no build, no npm dependencies, `node --test`.

```js
    //      roomId, runId, note, repeat, repeatUntil, worker, utility, periodFrom, periodTo } are checked by server/costs.js. ----

    // A new cost made at `at` (ms). → the cost.
    insertCost(fields, at) {
      return this.getCost(Number(costQ.insert.run(...costValues(fields), at).lastInsertRowid));
    },

    // A cost, or null.
    getCost(id) {
      const r = costQ.get.get(id);
      return r ? costOf(r) : null;
    },

    // Every cost, oldest date first.
    listCosts() {
      return costQ.list.all().map(costOf);
    },

    // → the cost with its new fields, or null when there is no such cost.
    updateCost(id, fields) {
      return costQ.update.run(...costValues(fields), id).changes > 0 ? this.getCost(id) : null;
    },

    // A monthly item's last month ('YYYY-MM'). → the cost, or null.
    stopCost(id, until) {
      return costQ.until.run(until, id).changes > 0 ? this.getCost(id) : null;
    },

    // A monthly item changed from a month: it ends at `until` and `fields` start a new one, in one transaction.
    // → { ended, started }, or null when there is no such cost (nothing changes then).
    changeCost(id, until, fields, at) {
      let started = null;
      inTransaction(() => {
        if (costQ.until.run(until, id).changes === 0) return;
        started = Number(costQ.insert.run(...costValues(fields), at).lastInsertRowid);
      });
      return started === null ? null : { ended: this.getCost(id), started: this.getCost(started) };
    },

    // → true when a cost went.
    deleteCost(id) {
      return costQ.delete.run(id).changes > 0;
    },
```

```js
const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

export const monthOf = (date) => date.slice(0, 7);
export const firstDayOf = (month) => `${month}-01`;
export function lastDayOf(month) {
  const [y, m] = month.split('-').map(Number);
  return new Date(Date.UTC(y, m, 0)).toISOString().slice(0, 10);
}
export function addMonths(month, n) {
  const [y, m] = month.split('-').map(Number);
  return new Date(Date.UTC(y, m - 1 + n, 1)).toISOString().slice(0, 7);
}
export const monthName = (month) => `${MONTHS[Number(month.slice(5)) - 1]} ${month.slice(0, 4)}`; // "Nov 2026"
export const dayMonth = (date) => `${Number(date.slice(8))} ${MONTHS[Number(date.slice(5, 7)) - 1]}`; // "3 Oct"
export const isBill = (cost) => cost.category === 'utilities' && BILLS.includes(cost.utility);

// The day a monthly item charges in `month`: the day of its date, or the month's last day when the month is shorter.
export function chargeDate(date, month) {
  const day = Math.min(Number(date.slice(8)), Number(lastDayOf(month).slice(8)));
  return `${month}-${String(day).padStart(2, '0')}`;
}

// A cost's charges in [from, to], none after today: [{ cost, date }]. A one-off charges on its date; a monthly item once a
// month from its date's month through repeatUntil (or on).
export function chargesOf(cost, from, to, today) {
  const end = to < today ? to : today;
  if (cost.repeat !== 'monthly') return cost.date >= from && cost.date <= end ? [{ cost, date: cost.date }] : [];
  const out = [];
  const last = cost.repeatUntil && cost.repeatUntil < monthOf(end) ? cost.repeatUntil : monthOf(end);
  for (let m = monthOf(cost.date > from ? cost.date : from); m <= last; m = addMonths(m, 1)) {
    const date = chargeDate(cost.date, m);
    if (date >= from && date <= end && date >= cost.date) out.push({ cost, date });
  }
  return out;
}

// A run is { id, roomId, startDate, endDate (null while open), blocks }: an open run reaches today.
export const runEnd = (run, today) => run.endDate ?? today;

// The room's run open on `date`: of those whose days hold it, the one started last (the day one ends and the next starts is
// the new one's), or null.
export function runOn(runs, roomId, date, today) {
  let found = null;
  for (const r of runs) {
    if (r.roomId !== roomId || r.startDate > date || date > runEnd(r, today)) continue;
    if (!found || r.startDate > found.startDate || (r.startDate === found.startDate && r.id > found.id)) found = r;
  }
  return found;
}

// The run's blocks × its days open in [from, to], its first and last day both counted.
export function blockDays(run, from, to, today) {
  const a = run.startDate > from ? run.startDate : from;
  const end = runEnd(run, today);
  const b = end < to ? end : to;
  return b < a ? 0 : run.blocks * (daysBetween(a, b) + 1);
}

// `amount` split over the weights to 2 places, the last taking the remainder so the shares sum to it.
export function split(amount, weights) {
  const total = weights.reduce((s, w) => s + w, 0);
  let left = amount;
  return weights.map((w, i) => {
    if (i === weights.length - 1) return round(left, 2);
    const share = round((amount * w) / total, 2);
    left -= share;
    return share;
  });
}
```

```js
// The measured kWh and litres of every room over [from, to] (bills' periods may reach before the period on show).
function measuredOver(from, to, data) {
  let kWh = 0;
  let waterL = 0;
  let pending = false;
  for (const days of data.usage.values()) {
    for (let d = from; d <= to; d = addDays(d, 1)) {
      const day = days.get(d);
      if (!day) continue;
      if (day.pending) pending = true;
      else {
        kWh += day.kWh;
        waterL += day.waterL;
      }
    }
  }
  return { kWh: round(kWh, 3), waterL: round(waterL, 3), pending };
}

// A bill's top-up (section 4): the bill − the measured cost of that utility over its period, shared when over 0; a note when
// the measured cost is higher, or when the bill counts in full (days not kept, no water price).
function topUp(bill, data) {
  const what = WHAT[bill.utility];
  const line = (amount) => ({
    key: `unmeasured-${bill.utility}`, category: 'utilities', label: `Unmeasured ${what}`, source: 'bill', amount,
    roomId: null, runId: null, shared: true, costId: bill.id,
  });
  if (bill.periodFrom < data.keptFrom) {
    return { line: line(bill.amount), note: note('bill-not-kept', `The ${what} bill of ${dayMonth(bill.date)} covers days before the energy kept: all of it counts`) };
  }
  if (bill.utility === 'water' && data.prices.water === null) {
    return { line: line(bill.amount), note: note('no-water-price', 'No water price is set: the water bill counts in full') };
  }
  const m = measuredOver(bill.periodFrom, bill.periodTo, data);
  if (m.pending) return { pending: true };
  const measured = bill.utility === 'electricity' ? round(m.kWh * data.prices.electricity, 2) : round((m.waterL / 1000) * data.prices.water, 2);
  const diff = round(bill.amount - measured, 2);
  if (diff > 0) return { line: line(diff) };
  return { note: note('bill-below', `Measured ${rm(measured)} is above the ${what} bill of ${rm(bill.amount)}: the estimate runs high`) };
}

// Each room's measured electricity and water over [from, to], one line per room and run (null: no run that day).
function measuredLines(from, to, data) {
  const lines = [];
  let pending = false;
  let notKept = false;
  for (const [roomId, days] of data.usage) {
    const groups = new Map();
    for (let d = from; d <= to; d = addDays(d, 1)) {
      if (d < data.keptFrom) {
        notKept = true;
        continue;
      }
      const day = days.get(d);
      if (!day) continue;
      if (day.pending) {
        pending = true;
        continue;
      }
      const runId = runOn(data.runs, roomId, d, data.today)?.id ?? null;
      const g = groups.get(runId) ?? { kWh: 0, waterL: 0 };
      g.kWh += day.kWh;
      g.waterL += day.waterL;
      groups.set(runId, g);
    }
    for (const [runId, g] of groups) {
      const at = { roomId, runId, shared: false };
      lines.push({ key: 'electricity', category: 'utilities', label: 'Electricity', source: 'measured', amount: round(round(g.kWh, 3) * data.prices.electricity, 2), ...at });
      if (data.prices.water !== null) {
        lines.push({ key: 'water', category: 'utilities', label: 'Water', source: 'measured', amount: round((round(g.waterL, 3) / 1000) * data.prices.water, 2), ...at });
      }
    }
  }
  const notes = notKept ? [note('energy-not-kept', `Energy before ${dayMonth(data.keptFrom)} ${data.keptFrom.slice(0, 4)} is not kept: not counted`)] : [];
  return { lines, notes, pending };
}

// Every cost line of [from, to]: { key, category, label, source ('manual' | 'batches' | 'inventory' | 'measured' | 'bill'),
// amount, roomId, runId, shared, date?, costId? }, with the notes of what is missing and pending (a day not worked out).
export function linesOf(from, to, data) {
  const lines = [];
  const notes = [];
  let pending = false;
  const roomOf = (id) => (id !== null && data.rooms.has(id) ? id : null);
  for (const c of data.costs) {
    for (const { date } of chargesOf(c, from, to, data.today)) {
      if (isBill(c)) {
        const t = topUp(c, data);
        if (t.pending) pending = true;
        if (t.line) lines.push({ ...t.line, date });
        if (t.note) notes.push(t.note);
        continue;
      }
      const roomId = roomOf(c.roomId);
      const runId = roomId === null ? null : c.repeat === 'monthly' ? (runOn(data.runs, roomId, date, data.today)?.id ?? null) : c.runId;
      lines.push({ key: c.category, category: c.category, label: CATEGORY_LABELS[c.category], source: 'manual', amount: c.amount, date, roomId, runId, shared: roomId === null, costId: c.id });
    }
  }
  for (const b of data.batches) {
    if (b.date < from || b.date > to) continue;
    const roomId = roomOf(b.roomId);
    if (b.cost === null) {
      notes.push(note('unpriced-batch', `${data.rooms.get(roomId) ?? 'A batch'}’s blocks of ${dayMonth(b.date)} have no price`, b.runId));
      continue;
    }
    lines.push({ key: 'blocks-batches', category: 'blocks', label: 'Blocks', source: 'batches', amount: b.cost, date: b.date, roomId, runId: b.runId, shared: roomId === null && b.runId === null });
  }
  for (const p of data.purchases) {
    if (p.date < from || p.date > to) continue;
    if (p.total === null) {
      notes.push(note('no-unit-cost', `${p.name} bought ${dayMonth(p.date)} has no unit cost`));
      continue;
    }
    const [category, label] = INVENTORY_LINES[p.kind] ?? INVENTORY_LINES.other;
    lines.push({ key: `inventory-${p.kind in INVENTORY_LINES ? p.kind : 'other'}`, category, label, source: 'inventory', amount: p.total, date: p.date, roomId: null, runId: null, shared: true });
  }
  const m = measuredLines(from, to, data);
  if (m.pending) pending = true;
  return { lines: [...lines, ...m.lines], notes: [...notes, ...m.notes], pending };
}

// The lines grouped by key: [{ key, label, source, category, amount, share (whole % of the costs) }], biggest first.
function categoryLines(lines, total) {
  const by = new Map();
  for (const l of lines) {
    const g = by.get(l.key) ?? { key: l.key, label: l.label, source: l.source, category: l.category, amount: 0 };
    g.amount = round(g.amount + l.amount, 2);
    by.set(l.key, g);
  }
  return [...by.values()].filter((g) => g.amount > 0) // a room that used nothing: no 0 line
    .map((g) => ({ ...g, share: total > 0 ? Math.round((g.amount / total) * 100) : null }))
    .sort((a, b) => b.amount - a.amount);
}

// What was picked and not yet sold: { kg, value (at the crop prices, null when none is priced) }, or null.
function unsoldOf(harvests, picked, sold, cropYield) {
  if (!(picked > sold)) return null;
  const kg = round(picked - sold, 2);
  const value = valueOf(harvests, cropYield);
  return { kg, value: value === null ? null : round((value * kg) / picked, 2) };
}

const PARTS = ['blocks', 'energy', 'water', 'own', 'shared'];
const emptyPart = () => ({ blocks: 0, energy: 0, water: 0, own: 0, shared: 0, income: 0, kgHarvested: 0, kgSold: 0, byCategory: {} });
const partOf = (l) => (l.source === 'batches' ? 'blocks' : l.source === 'measured' ? (l.key === 'electricity' ? 'energy' : 'water') : 'own');

// The figures of [from, to] (section 4): { from, to, lines, byCategory, income, costs, profit, kgHarvested, kgSold, costPerKg,
// profitPerKg, unsold, runs (Map run id → { blocks, energy, water, own, shared, income, kgHarvested, kgSold, byCategory }),
// notes [{ kind, text, runId? }], pending }. While pending: costs, profit and the per-kg figures null.
export function figuresOf(from, to, data) {
  const { lines, notes, pending } = linesOf(from, to, data);
  const runs = new Map();
  const part = (id) => {
    if (!runs.has(id)) runs.set(id, emptyPart());
    return runs.get(id);
  };
  for (const l of lines) {
    if (l.shared || l.runId === null) continue;
    const p = part(l.runId);
    const k = partOf(l);
    p[k] = round(p[k] + l.amount, 2);
    if (k !== 'energy' && k !== 'water') p.byCategory[l.category] = round((p.byCategory[l.category] ?? 0) + l.amount, 2);
  }
  const sharedTotal = sum(lines.filter((l) => l.shared), 'amount');
  const open = data.runs.map((r) => ({ id: r.id, w: blockDays(r, from, to, data.today) })).filter((x) => x.w > 0);
  if (sharedTotal > 0 && open.length === 0) notes.push(note('not-shared', `${rm(sharedTotal)} of shared costs not shared: no blocks in`));
  if (sharedTotal > 0 && open.length > 0) {
    split(sharedTotal, open.map((x) => x.w)).forEach((amount, i) => {
      const p = part(open[i].id);
      p.shared = round(p.shared + amount, 2);
    });
  }
  const sales = data.sales.filter((s) => s.date >= from && s.date <= to);
  for (const s of sales) {
    const runId = s.runId ?? (s.roomId === null ? null : runOn(data.runs, s.roomId, s.date, data.today)?.id ?? null);
    if (runId === null) continue;
    const p = part(runId);
    p.income = round(p.income + s.total, 2);
    p.kgSold = round(p.kgSold + s.kg, 2);
  }
  const harvests = data.harvests.filter((h) => h.date >= from && h.date <= to);
  for (const h of harvests) {
    if (h.runId === null) continue;
    const p = part(h.runId);
    p.kgHarvested = round(p.kgHarvested + h.kg, 2);
  }
  const costs = sum(lines, 'amount');
  const income = sum(sales, 'total');
  const kgSold = sum(sales, 'kg');
  const kgHarvested = sum(harvests, 'kg');
  const profit = round(income - costs, 2);
  const unsold = unsoldOf(harvests, kgHarvested, kgSold, data.cropYield);
  if (unsold) {
    const about = unsold.value === null ? '' : ` (about ${rm(unsold.value)} at your prices)`;
    notes.unshift(note('unsold', `${unsold.kg.toFixed(1)} kg picked and not yet sold${about}`));
  }
  if (sales.length === 0) notes.push(note('no-sales', 'No sales recorded'));
  if (data.prices.water === null && !notes.some((n) => n.kind === 'no-water-price')) notes.push(note('no-water-price', 'No water price is set: energy is electricity only'));
  return {
    from, to, lines, byCategory: categoryLines(lines, costs), income, kgHarvested, kgSold, unsold, runs,
    costs: pending ? null : costs,
    profit: pending ? null : profit,
    costPerKg: pending || kgHarvested === 0 ? null : costPerKgOf(costs, kgHarvested),
    profitPerKg: pending || kgSold === 0 ? null : costPerKgOf(profit, kgSold),
    notes: notes.filter((n, i) => notes.findIndex((x) => x.text === n.text) === i),
    pending,
  };
}

// A run's whole figures: its parts summed over every month from its start to today (sales and costs tied to it may come after
// it ends). figs(month) → figuresOf's result for that month. → { parts, byCategory [{ category, amount }] ('shared' last),
// income, kgHarvested, kgSold, costs, profit, costPerKg, profitPerKg, notes, pending }.
export function runTotals(run, figs, today) {
  const t = emptyPart();
  const notes = [];
  let pending = false;
  for (let m = monthOf(run.startDate); m <= monthOf(today); m = addMonths(m, 1)) {
    const f = figs(m);
    pending ||= f.pending;
    notes.push(...f.notes.filter((n) => n.runId === run.id || (n.kind === 'energy-not-kept' && m === monthOf(run.startDate))));
    const p = f.runs.get(run.id);
    if (!p) continue;
    for (const k of [...PARTS, 'income', 'kgHarvested', 'kgSold']) t[k] = round(t[k] + p[k], 2);
    for (const [c, v] of Object.entries(p.byCategory)) t.byCategory[c] = round((t.byCategory[c] ?? 0) + v, 2);
  }
  const costs = round(PARTS.reduce((s, k) => s + t[k], 0), 2);
  const profit = round(t.income - costs, 2);
  const byCategory = [
    ...CATEGORIES.filter((c) => t.byCategory[c]).map((c) => ({ category: c, amount: t.byCategory[c] })),
    ...(t.shared > 0 ? [{ category: 'shared', amount: t.shared }] : []),
  ];
  return {
    parts: Object.fromEntries(PARTS.map((k) => [k, t[k]])), byCategory, income: t.income, kgHarvested: t.kgHarvested,
    kgSold: t.kgSold,
    costs: pending ? null : costs,
    profit: pending ? null : profit,
    costPerKg: pending || t.kgHarvested === 0 ? null : costPerKgOf(costs, t.kgHarvested),
    profitPerKg: pending || t.kgSold === 0 ? null : costPerKgOf(profit, t.kgSold),
    notes: notes.filter((n, i) => notes.findIndex((x) => x.text === n.text) === i),
    pending,
  };
}

// The contract's summary (GET /api/costs/summary): { total, byCategory [{ category, amount }] (the measured energy and water
// apart), energy, water, income, profit, profitPerKg }.
export function summaryOf(f) {
  const by = {};
  for (const l of f.lines) if (l.source !== 'measured') by[l.category] = round((by[l.category] ?? 0) + l.amount, 2);
  const measured = (key) => sum(f.lines.filter((l) => l.source === 'measured' && l.key === key), 'amount');
  return {
    total: f.costs, byCategory: CATEGORIES.filter((c) => by[c]).map((c) => ({ category: c, amount: by[c] })),
    energy: measured('electricity'), water: measured('water'), income: f.income, profit: f.profit, profitPerKg: f.profitPerKg,
  };
}
```

```js
  && new Date(`${s}T00:00:00Z`).toISOString().slice(0, 10) === s;

export function createCosts({ db, now = Date.now, zone = localZone, log = console.error }) {
  const listeners = new Set();
  function tell() {
    for (const fn of [...listeners]) {
      try {
        fn();
      } catch (err) {
        log(err);
      }
    }
  }

  const today = () => zone.label(now());
  const keptFrom = () => addDays(today(), -(USAGE_DAYS_KEPT - 1));

  // Every room's runs as costs-math wants them (dates in the usage days' zone), with what the page shows.
  function runsOf() {
    return db.listRooms().flatMap((room) => db.cropRuns(room.id).map((r) => ({
      id: r.id, roomId: r.roomId, roomName: room.name, startedAt: r.startedAt, endedAt: r.endedAt, crops: r.crops,
      startDate: zone.label(r.startedAt), endDate: r.endedAt === null ? null : zone.label(r.endedAt),
      blocks: r.crops.reduce((s, c) => s + c.blocks, 0),
    })));
  }

  // Who was paid: the tasks' rule (an active worker when the farm has a list).
  function workerOf(worker) {
    try {
      return checkWorker(db, worker);
    } catch (err) {
      if (err instanceof TaskError) throw new CostError(err.status, err.message, 'worker');
      throw err;
    }
  }

  // A cost's fields from a body (section 2); old: the cost an edit changes (a field left out keeps its value; repeat and
  // repeatUntil are changed only by change and stop).
  function fieldsOf(body, old = null) {
    const b = bodyOf(body);
    const pick = (k) => (b[k] !== undefined ? b[k] : old?.[k]);
    const t = today();
    const date = pick('date');
    if (!isDate(date) || date > t) throw new CostError(400, 'Pick a date up to today', 'date');
    const category = pick('category');
    if (!CATEGORIES.includes(category)) throw new CostError(400, 'Pick a category', 'category');
    const given = pick('amount');
    const amount = typeof given === 'number' && Number.isFinite(given) ? Math.round(given * 100) / 100 : NaN;
    if (!(amount > 0 && amount <= MAX_AMOUNT)) throw new CostError(400, 'RM must be a number over 0 up to 1,000,000', 'amount');
    const roomId = pick('roomId') ?? null;
    if (roomId !== null && !(Number.isInteger(roomId) && db.getRoom(roomId))) throw new CostError(400, 'That container no longer exists', 'roomId');
    const text = pick('note') ?? '';
    if (typeof text !== 'string') throw new CostError(400, 'The note must be text', 'note');
    const note = text.trim();
    if (note.length > MAX_NOTE) throw new CostError(400, `The note must be at most ${MAX_NOTE} characters`, 'note');
    const repeat = old ? old.repeat : (b.repeat ?? null);
    if (repeat !== null && repeat !== 'monthly') throw new CostError(400, 'Repeat must be monthly or none', 'repeat');
    const repeatUntil = old?.repeatUntil ?? null;
    if (repeatUntil !== null && repeatUntil < monthOf(date)) throw new CostError(400, `Pick a date up to ${monthName(repeatUntil)}, its last month`, 'date');
    const utility = category === 'utilities' ? pick('utility') : null;
    if (category === 'utilities' && !UTILITIES.includes(utility)) throw new CostError(400, 'Pick electricity, water or other', 'utility');
    const bill = category === 'utilities' && BILLS.includes(utility);
    if (bill && repeat) throw new CostError(400, 'An electricity or water bill is entered each month', 'repeat');
    if (bill && roomId !== null) throw new CostError(400, 'A bill is the whole farm’s', 'roomId');
    const worker = category === 'labour' ? workerOf(pick('worker')) : null;
    let periodFrom = null;
    let periodTo = null;
    if (bill) {
      periodFrom = pick('periodFrom') ?? firstDayOf(monthOf(date));
      periodTo = pick('periodTo') ?? lastDayOf(monthOf(date));
      if (!isDate(periodFrom)) throw new CostError(400, 'Pick the first day the bill covers', 'periodFrom');
      if (!isDate(periodTo) || periodTo < periodFrom || daysBetween(periodFrom, periodTo) + 1 > MAX_BILL_DAYS) {
        throw new CostError(400, `Pick the last day the bill covers, up to ${MAX_BILL_DAYS} days after the first`, 'periodTo');
      }
    }
    // A one-off room cost: the run open in the room on its date. A monthly item finds each charge's run on read.
    const runId = roomId === null || repeat ? null : (runOn(runsOf(), roomId, date, t)?.id ?? null);
    return { date, category, amount, roomId, runId, note, repeat, repeatUntil, worker, utility, periodFrom, periodTo };
  }

  function costOf(id) {
    const c = db.getCost(id);
    if (!c) throw new CostError(404, 'Cost not found');
    return c;
  }
  function monthlyOf(id) {
    const c = costOf(id);
    if (c.repeat !== 'monthly') throw new CostError(400, 'This cost is not a monthly item');
    return c;
  }
  // `from` (change): its first month (edited in place) to this month, or to its last month when stopped earlier. `after`
  // (stop): its first month to this month.
  function monthIn(value, c, field) {
    const lowest = monthOf(c.date);
    const current = monthOf(today());
    const last = field === 'from' && c.repeatUntil && c.repeatUntil < current ? c.repeatUntil : current;
    if (!isMonth(value) || value < lowest || value > last) throw new CostError(400, `Pick a month from ${monthName(lowest)} to ${monthName(last)}`, field);
    return value;
  }

  return {
    // Every cost (the edit dialogs).
    list() {
      return db.listCosts();
    },

    add(body) {
      const made = db.insertCost(fieldsOf(body), now());
      tell();
      return made;
    },

    edit(id, body) {
      const row = db.updateCost(id, fieldsOf(body, costOf(id)));
      tell();
      return row;
    },

    remove(id) {
      if (!db.deleteCost(id)) throw new CostError(404, 'Cost not found');
      tell();
    },

    // A monthly item's new amount (and note) from a month (section 2.1): from its first month, edited in place ({ ended:
    // null, started: the item }); later, it ends the month before and a new one starts. → { ended, started }.
    change(id, body) {
      const c = monthlyOf(id);
      const b = bodyOf(body);
      const from = monthIn(b.from, c, 'from');
      const fields = fieldsOf({ amount: b.amount, note: b.note ?? c.note }, c);
      if (from === monthOf(c.date)) {
        const started = db.updateCost(id, fields);
        tell();
        return { ended: null, started };
      }
      const changed = db.changeCost(id, addMonths(from, -1), { ...fields, date: chargeDate(c.date, from) }, now());
      tell();
      return changed;
    },

    // A monthly item's last month (section 2.1). → the item.
    stopItem(id, body) {
      const c = monthlyOf(id);
      const after = monthIn(bodyOf(body).after, c, 'after');
      const row = db.stopCost(id, after);
      tell();
      return row;
    },

    // fn() after any write here or a usage warm done (the figures can be worked out now); → a function that stops it.
    onChange(fn) {
      listeners.add(fn);
      return () => listeners.delete(fn);
    },
  };
}
```

```js
  // the request (a year of one-second readings takes seconds); it starts a warm, one at a time a room, that tells when done.
  const warming = new Map();
  const stopping = new AbortController();
  function warm(roomId, count) {
    if (stopping.signal.aborted) return;
    const running = warming.get(roomId);
    if (running) {
      running.more = Math.max(running.more, count);
      return;

    const job = { count, more: 0 };

        if (!stopping.signal.aborted) tell();
      } catch (err) {
        warming.delete(roomId);
```

```js
// The costs routes (spec 2026-11-05-costs-design.md 5), called from api.js: GET /api/costs, /api/costs/month?month=,
// /api/costs/summary?from=&to=&run=; POST /api/costs; PUT and DELETE /api/costs/:id; POST /api/costs/:id/change and /stop.
// server/costs.js checks, writes and works out; this reads the query and the body and turns its errors into the API's.

const ONE = /^\/api\/costs\/([1-9][0-9]{0,14})$/;
const ACTION = /^\/api\/costs\/([1-9][0-9]{0,14})\/(change|stop)$/;

// costs: from createCosts. → async (req, res, path, { send, readJson, HttpError, url }) that answers true when the path is one
// of its routes.
export function createCostRoutes({ costs }) {
  async function route(req, res, path, { send, readJson, HttpError, url }) {
    const notFound = () => new HttpError(404, 'Not found');
    const q = (name) => url.searchParams.get(name);

    if (path === '/api/costs') {
      if (req.method === 'GET') send(res, 200, { costs: costs.list() });
      else if (req.method === 'POST') send(res, 201, costs.add(await readJson(req)));
      else throw notFound();
      return true;
    }
    if (path === '/api/costs/month') {
      if (req.method !== 'GET') throw notFound();
      send(res, 200, costs.monthView({ month: q('month') }));
      return true;
    }
    if (path === '/api/costs/summary') {
      if (req.method !== 'GET') throw notFound();
      const run = q('run');
      if (run !== null && !/^[1-9][0-9]{0,14}$/.test(run)) throw new HttpError(400, 'run must be a run id', 'run');
      send(res, 200, costs.summary({ from: q('from'), to: q('to'), run: run === null ? null : Number(run) }));
      return true;
    }
    const action = ACTION.exec(path);
    if (action) {
      if (req.method !== 'POST') throw notFound();
      const id = Number(action[1]);
      const body = await readJson(req);
      send(res, 200, action[2] === 'change' ? costs.change(id, body) : costs.stopItem(id, body));
      return true;
    }
    const one = ONE.exec(path);
    if (one) {
      const id = Number(one[1]);
      if (req.method === 'PUT') {
        send(res, 200, costs.edit(id, await readJson(req)));
      } else if (req.method === 'DELETE') {
        costs.remove(id);
        res.writeHead(204);
        res.end();
      } else {
        throw notFound();
      }
      return true;
    }
    return false;
  }

  // A 4xx from server/costs.js (CostError) becomes the API's HttpError.
  return async function costRoutes(req, res, path, helpers) {
    try {
      return await route(req, res, path, helpers);
    } catch (err) {
      const { HttpError } = helpers;
      if (!(err instanceof HttpError) && Number.isInteger(err?.status) && err.status >= 400 && err.status < 500) {
        throw new HttpError(err.status, err.message, err.field);
      }
      throw err;
    }
  };
}
```

```js
// The Costs page's pure parts (spec 2026-11-05-costs-design.md 6): the one formatter for RM (a loss with a leading "−", the
// One Format Rule), the month words, the month tiles, the rows of By category, Entries, Monthly items and Runs, a run's
// breakdown and the 12 months' profit bars. Every formatter shows a dash for a missing figure. A number keeps its unit at a
// wrap (a no-break space, the Units Rule); kg and cost per kg come from Harvest's formatter.

import { kgText, costKgText } from './harvest-math.js';
import { cropShortName } from './crops.js';
import { DOT } from './text-math.js';
import { niceScale } from './usage-math.js';

const NB = ' ';
const DASH = '—';
const MINUS = '−';
const MONTHS = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];
const FARM_OFFSET_MS = 8 * 3_600_000; // the farm's time is UTC+8 all year
const finite = (x) => typeof x === 'number' && Number.isFinite(x);
const grouped = (n, places) => n.toLocaleString('en-US', { minimumFractionDigits: places, maximumFractionDigits: places });

// "RM 1,240.50", "−RM 1,240.50".
export function moneyText(v) {
  if (!finite(v)) return DASH;
  const t = `RM${NB}${grouped(Math.abs(v), 2)}`;
  return v < 0 ? `${MINUS}${t}` : t;
}
// A cost or profit per kg: Harvest's rule (never a false RM 0.00), a loss with a "−".
export const perKgText = (v) => (!finite(v) ? DASH : v < 0 ? `${MINUS}${costKgText(-v)}` : costKgText(v));
const axisMoney = (v) => `${v < 0 ? MINUS : ''}RM${NB}${grouped(Math.abs(v), 0)}`;

export const monthTitle = (month) => `${MONTHS[Number(month.slice(5)) - 1]} ${month.slice(0, 4)}`; // "November 2026"
const monthShort = (month) => `${MONTHS[Number(month.slice(5)) - 1].slice(0, 3)} ${month.slice(0, 4)}`; // "Nov 2026"
export function addMonths(month, n) {
  const [y, m] = month.split('-').map(Number);
  return new Date(Date.UTC(y, m - 1 + n, 1)).toISOString().slice(0, 7);
}
export function monthsBetween(a, b) {
  const out = [];
  for (let m = a; m <= b; m = addMonths(m, 1)) out.push(m);
  return out;
}
const dayText = (date) => `${Number(date.slice(8))} ${MONTHS[Number(date.slice(5, 7)) - 1].slice(0, 3)}`; // "3 Nov"
const farmDay = (ms) => new Date(ms + FARM_OFFSET_MS).toISOString().slice(0, 10);

const PENDING = 'Working out the energy cost…';

// The month card's tiles (harvest-card.js's tileNodes): [{ label, value, line? }].
export function monthTiles({ pending, tiles: t }) {
  const sold = t.kgSold > 0 ? `${kgText(t.kgSold)} sold` : 'Nothing sold';
  return [
    { label: 'Income', value: moneyText(t.income), line: sold },
    pending ? { label: 'Costs', value: DASH, line: PENDING } : { label: 'Costs', value: moneyText(t.costs) },
    pending ? { label: 'Profit', value: DASH, line: PENDING } : { label: 'Profit', value: moneyText(t.profit), ...(t.profit < 0 ? { line: 'Loss' } : {}) },
    { label: 'Cost per kg', value: perKgText(t.costPerKg), line: t.kgHarvested > 0 ? `${kgText(t.kgHarvested)} picked` : 'Nothing picked' },
    { label: 'Profit per kg', value: perKgText(t.profitPerKg), line: t.kgSold > 0 ? 'per kg sold' : 'Nothing sold yet' },
  ];
}

// Where a line comes from, in a muted line under its name (nothing for one entered here).
export const SOURCE_WORDS = { manual: null, batches: 'from Batches', inventory: 'from Inventory', measured: 'measured', bill: `bill${NB}−${NB}measured` };
export const categoryRows = (list) => list.map((c) => ({
  label: c.label, source: SOURCE_WORDS[c.source] ?? null, amount: moneyText(c.amount), share: finite(c.share) ? `${c.share}${NB}%` : DASH,
}));

const LABELS = {
  blocks: 'Blocks', spawn: 'Spawn', labour: 'Labour', rent: 'Rent', repairs: 'Repairs', transport: 'Transport',
  packaging: 'Packaging', utilities: 'Utilities', other: 'Other',
};
const UTILITY_WORDS = { electricity: 'Electricity bill', water: 'Water bill', other: null };
const whatOf = (c) => [c.utility ? UTILITY_WORDS[c.utility] ?? LABELS[c.category] : LABELS[c.category], c.worker, c.note].filter(Boolean).join(DOT);

// An entry of the month: { date, what, whom, amount, monthly }.
export const entryLine = (e) => ({
  date: dayText(e.date), what: whatOf(e), whom: e.roomName ?? 'The farm', amount: moneyText(e.amount), monthly: e.repeat === 'monthly',
});

// A monthly item: { what, whom, amount, since }.
export const repeatingLine = (c) => ({
  what: whatOf(c), whom: c.roomName ?? 'The farm', amount: `${moneyText(c.amount)} a${NB}month`, since: `since ${monthShort(c.date.slice(0, 7))}`,
});

export const RUN_COLUMNS = ['Dates', 'Container', 'Crops', 'Income', 'Costs', 'Profit', 'Profit per kg'];

// A run's row: { dates, room, crops, cells (Income, Costs, Profit, Profit per kg), loss }.
export function runRow(run) {
  const to = run.endedAt === null ? 'now' : dayText(farmDay(run.endedAt));
  return {
    dates: `${dayText(farmDay(run.startedAt))}${NB}–${NB}${to}`, room: run.roomName,
    crops: run.crops.map((c) => cropShortName(c.crop)).join(', '),
    cells: [moneyText(run.income), moneyText(run.costs), moneyText(run.profit), perKgText(run.profitPerKg)],
    loss: finite(run.profit) && run.profit < 0,
  };
}

// A run's breakdown under its row: [[label, value]], the parts that are not 0, then income.
export function runParts(run) {
  const p = run.parts;
  return [
    ['Blocks', p.blocks], ['Electricity', p.energy], ['Water', p.water], ['Own costs', p.own], ['Share of shared costs', p.shared],
  ].filter(([, v]) => v > 0).map(([label, v]) => [label, moneyText(v)]).concat([['Income', moneyText(run.income)]]);
}

export const BARS_M = { left: 72, right: 8, top: 10, bottom: 22 };

// The 12 months' profit bars in a width × height box: { bars: [{ month, label, x, y, w, h, loss, pending, shown }], zeroY,
// ticks: [{ y, label }] }. Profits rise from the zero line, losses hang from it; each side scaled by niceScale.
export function profitBars(months, shown, width, height, m = BARS_M) {
  const values = months.map((x) => (finite(x.profit) ? x.profit : 0));
  const hi = Math.max(0, ...values);
  const lo = Math.min(0, ...values);
  const up = hi > 0 ? niceScale(hi, 4) : null;
  const down = lo < 0 ? niceScale(-lo, 4) : null;
  const top = up?.max ?? (down ? 0 : 1);
  const bottom = down?.max ?? 0;
  const plotH = height - m.top - m.bottom;
  const y = (v) => m.top + ((top - v) / (top + bottom)) * plotH;
  const slot = (width - m.left - m.right) / Math.max(1, months.length);
  const w = Math.max(2, slot * 0.64);
  const bars = months.map((x, i) => {
    const v = values[i];
    const [y0, y1] = [y(0), y(v)];
    return {
      month: x.month, label: MONTHS[Number(x.month.slice(5)) - 1].slice(0, 3), x: m.left + i * slot + (slot - w) / 2,
      y: Math.min(y0, y1), w, h: Math.max(1, Math.abs(y1 - y0)), loss: v < 0, pending: x.profit === null, shown: x.month === shown,
    };
  });
  const values2 = [...(up?.ticks ?? [0]), ...(down?.ticks.slice(1).map((v) => -v) ?? [])];
  return { bars, zeroY: y(0), ticks: values2.map((v) => ({ y: y(v), label: axisMoney(v) })) };
}
```

```js
// The Costs page's dialogs (spec 2026-11-05-costs-design.md 6), through the tasks' openForm: Add a cost, Edit a one-off (with
// a quiet Delete), Change a monthly item from a month (with its Delete), Stop it, and the Delete confirm. Each sends one
// request and calls onDone() when it went through; the page reloads on the costs event too.

import { request } from './api.js';
import { openForm, loadWorkers, workerField, errorText } from './task-dialogs.js';
import { moneyText, monthTitle, monthsBetween } from './costs-math.js';

const CATEGORY_OPTIONS = [
  ['labour', 'Labour'], ['rent', 'Rent'], ['utilities', 'Utilities'], ['repairs', 'Repairs'], ['transport', 'Transport'],
  ['packaging', 'Packaging'], ['blocks', 'Blocks'], ['spawn', 'Spawn'], ['other', 'Other'],
];
const UTILITY_OPTIONS = [['electricity', 'Electricity bill'], ['water', 'Water bill'], ['other', 'Other (internet, fixed charges)']];
const isBill = (v) => v.category === 'utilities' && (v.utility === 'electricity' || v.utility === 'water');

// A refused send's words on its box, or the dialog's line.
const problem = (err) => (err.field ? { field: err.field, message: err.message } : { message: errorText(err) });
const amountOf = (text) => Number(String(text).replace(/,/g, '').trim() || NaN);

// The body of Add and Edit from the dialog's values.
function bodyOf(v) {
  const body = { date: v.date, category: v.category, amount: amountOf(v.amount), note: v.note };
  body.roomId = v.roomId ? Number(v.roomId) : null;
  if (v.category === 'labour') body.worker = v.worker;
  if (v.category === 'utilities') body.utility = v.utility;
  if (isBill(v)) {
    body.roomId = null;
    if (v.periodFrom) body.periodFrom = v.periodFrom;
    if (v.periodTo) body.periodTo = v.periodTo;
  }
  return body;
}

// Shows only the boxes the category needs: the worker for labour, the kind for utilities, the period for a bill (whose
// "For" is the farm's).
function followCategory(form) {
  const show = (name, on) => { form.input(name).closest('.field').hidden = !on; };
  const sync = () => {
    const category = form.input('category').value;
    const utility = form.input('utility').value;
    const bill = category === 'utilities' && (utility === 'electricity' || utility === 'water');
    show('worker', category === 'labour');
    show('utility', category === 'utilities');
    show('periodFrom', bill);
    show('periodTo', bill);
    show('roomId', !bill);
    if (form.input('repeat')) show('repeat', !bill);
  };
  form.input('category').addEventListener('change', sync);
  form.input('utility').addEventListener('change', sync);
  sync();
}

// The shared boxes of Add and Edit. c: the cost (Edit) or null; rooms: [{ id, name }]; today: 'YYYY-MM-DD'.
function costFields(c, { rooms, workers, today }) {
  return [
    { name: 'date', label: 'Date', kind: 'date', value: c?.date ?? today, max: today },
    { name: 'category', label: 'Category', kind: 'select', value: c?.category ?? 'labour', options: CATEGORY_OPTIONS },
    { name: 'amount', label: 'Amount', kind: 'number', value: c ? String(c.amount) : '', suffix: 'RM' },
    { name: 'roomId', label: 'For', kind: 'select', value: c?.roomId ? String(c.roomId) : '', options: [['', 'The farm (shared)'], ...rooms.map((r) => [String(r.id), r.name])] },
    { ...workerField(workers, 'Worker'), value: c?.worker ?? workerField(workers).value },
    { name: 'utility', label: 'Kind', kind: 'select', value: c?.utility ?? 'electricity', options: UTILITY_OPTIONS },
    { name: 'periodFrom', label: 'Bill from', kind: 'date', value: c?.periodFrom ?? '' , hint: 'Empty: the first of the bill date’s month' },
    { name: 'periodTo', label: 'Bill to', kind: 'date', value: c?.periodTo ?? '', hint: 'Empty: the last of the bill date’s month' },
    { name: 'note', label: 'Note', kind: 'text', value: c?.note ?? '', maxLength: 200, placeholder: 'What it was for' },
  ];
}

// Add a cost.
export async function openAddDialog({ rooms, today, onDone, onLost }) {
  const workers = await loadWorkers().catch(() => []);
  const form = openForm({
    title: 'Add a cost',
    fields: [...costFields(null, { rooms, workers, today }), { name: 'repeat', label: 'Every month (rent, a base wage)', kind: 'check', value: false }],
    buttons: [{ value: 'add', label: 'Add', primary: true }],
    onLost,
    async submit(v) {
      try {
        await request('POST', '/api/costs', { ...bodyOf(v), repeat: v.repeat && !isBill(v) ? 'monthly' : null });
        onDone();
        return true;
      } catch (err) {
        return problem(err);
      }
    },
  });
  followCategory(form);
}

// Edit a one-off, with a quiet Delete at the footer's start.
export async function openEditDialog({ cost, rooms, today, onDone, onLost }) {
  const workers = await loadWorkers().catch(() => []);
  const form = openForm({
    title: 'Edit a cost',
    fields: costFields(cost, { rooms, workers, today }),
    buttons: [
      { label: 'Delete this cost', action: () => openDeleteDialog({ cost, onDone, onLost }) },
      { value: 'save', label: 'Save', primary: true },
    ],
    onLost,
    async submit(v) {
      try {
        await request('PUT', `/api/costs/${cost.id}`, bodyOf(v));
        onDone();
        return true;
      } catch (err) {
        return problem(err);
      }
    },
  });
  followCategory(form);
}

// A monthly item's new amount from a month (earlier months keep the old one), with its Delete.
export function openChangeDialog({ cost, month, onDone, onLost }) {
  const months = monthsBetween(cost.date.slice(0, 7), cost.repeatUntil && cost.repeatUntil < month ? cost.repeatUntil : month);
  openForm({
    title: `Change ${cost.note || 'this monthly item'}`,
    lead: [`${moneyText(cost.amount)} a month since ${monthTitle(cost.date.slice(0, 7))}. Earlier months keep what they were.`],
    fields: [
      { name: 'from', label: 'From', kind: 'select', value: months.at(-1), options: months.map((m) => [m, monthTitle(m)]) },
      { name: 'amount', label: 'Amount a month', kind: 'number', value: String(cost.amount), suffix: 'RM' },
      { name: 'note', label: 'Note', kind: 'text', value: cost.note, maxLength: 200 },
    ],
    buttons: [
      { label: 'Delete this item', action: () => openDeleteDialog({ cost, onDone, onLost }) },
      { value: 'save', label: 'Save', primary: true },
    ],
    onLost,
    async submit(v) {
      try {
        await request('POST', `/api/costs/${cost.id}/change`, { from: v.from, amount: amountOf(v.amount), note: v.note });
        onDone();
        return true;
      } catch (err) {
        return problem(err);
      }
    },
  });
}

// A monthly item's last month.
export function openStopDialog({ cost, month, onDone, onLost }) {
  const months = monthsBetween(cost.date.slice(0, 7), month);
  openForm({
    title: `Stop ${cost.note || 'this monthly item'}`,
    lead: ['It charges up to and including the month you pick. The months before keep their charges.'],
    fields: [{ name: 'after', label: 'Last month', kind: 'select', value: months.at(-1), options: months.map((m) => [m, monthTitle(m)]) }],
    buttons: [{ value: 'stop', label: 'Stop', primary: true }],
    onLost,
    async submit(v) {
      try {
        await request('POST', `/api/costs/${cost.id}/stop`, { after: v.after });
        onDone();
        return true;
      } catch (err) {
        return problem(err);
      }
    },
  });
}

// The Delete confirm: a monthly item goes with every month it charged.
export function openDeleteDialog({ cost, onDone, onLost }) {
  const monthly = cost.repeat === 'monthly';
  const what = cost.note || cost.category;
  openForm({
    title: monthly ? `Delete ${what}, ${moneyText(cost.amount)} a month?` : `Delete ${what}, ${moneyText(cost.amount)}?`,
    lead: [monthly ? `Every month it charged since ${monthTitle(cost.date.slice(0, 7))} goes too. To end it from a month, use Stop.` : 'This cannot be undone.'],
    buttons: [{ value: 'delete', label: 'Delete', danger: true, primary: true }],
    onLost,
    async submit() {
      try {
        await request('DELETE', `/api/costs/${cost.id}`);
        onDone();
        return true;
      } catch (err) {
        return problem(err);
      }
    },
  });
}
```

```js
function el(tag, className, content) {
  const node = document.createElement(tag);
  if (className) node.className = className;
  if (content !== undefined && content !== null) node.textContent = content;
  return node;
}
function svg(name, attrs = {}, text) {

const quietButton = (label, onClick) => {

let mountSeq = 0; // every mount gets a token; an answer that finds the token changed is dropped

export function mount(container, { month = null } = {}) {
  const me = ++mountSeq;
  const alive = () => me === mountSeq;
  container.innerHTML = TEMPLATE;
  const q = (sel) => container.querySelector(sel);
  const els = {

  document.title = 'Costs · Mushroom farm';

  let view = null; // GET /api/costs/month of the month on show
  let costs = []; // GET /api/costs: the rows the edit dialogs open
  let rooms = [];
  let seq = 0;
  let timer = null;
  const open = new Set(); // the runs whose breakdown is open

  function showError(text) {
    els.bannerText.textContent = text;
    els.banner.hidden = false;

  async function load() {
    const mine = ++seq;
    try {
      const [v, list, roomList] = await Promise.all([

  function paintHead() {
    const m = view.month;
    els.month.textContent = monthTitle(m);
    const prev = addMonths(m, -1);
    const next = addMonths(m, 1);
    const current = view.today.slice(0, 7);
    const step = (a, to, on) => {
      if (on) {
        a.href = costsHash(to === current ? null : to);
        a.removeAttribute('aria-disabled');
      } else {
        a.removeAttribute('href');
        a.setAttribute('aria-disabled', 'true');

    step(els.next, next, next <= current);

  function paintMonth() {
    els.title.textContent = monthTitle(view.month);
    els.tiles.replaceChildren(...tileNodes(monthTiles(view)));

    const width = Math.floor(els.chart.clientWidth) || 600;

      return;
    }
    const list = el('ul', 'co-entries');

        more.append(td);
        rows.push(more);
      }
    }
    els.runs.replaceChildren(table(`Runs in ${monthTitle(view.month)}`, RUN_COLUMNS, rows));
  }

  function paint() {
    els.body.hidden = false;
    paintHead();
    paintMonth();
    paintCategories();
    paintEntries();
    paintRepeating();
    paintRuns();
  }

  els.add.addEventListener('click', () => { if (view) openAddDialog(dialogOpts()); });
  els.retry.addEventListener('click', () => load());
  const resize = new ResizeObserver(() => { if (view) requestAnimationFrame(() => { if (alive() && view) paintChart(); }); });
  resize.observe(els.chart);
  const offLive = [
    ...LISTEN.map((type) => live.on(type, () => reload.call())),
    live.on('room', (d) => { if (d.part === 'crop' || d.part === 'room') reload.call(); }),
    live.onStatus((up) => {
      clearInterval(timer);
      if (up) reload.call();
      else timer = setInterval(() => reload.call(), REFRESH_MS);
    }),
  ];
  load();

  return () => {
    mountSeq++;
    reload.cancel();
    clearInterval(timer);
    resize.disconnect();
    for (const off of offLive) off();
  };
}
```
