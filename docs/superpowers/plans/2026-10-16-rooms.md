`server/rooms.js`:

```js
// Validation of room input. Pure: no database, no HTTP.

export const NAME_MAX = 60;

const SIZES = [
  { field: 'length', label: 'Length', min: 0.5, max: 100 },
  { field: 'width', label: 'Width', min: 0.5, max: 100 },
  { field: 'height', label: 'Height', min: 0.5, max: 10 },
];

// Returns { room: { name, length, width, height } } when the input is valid,
// otherwise { error, field } for the first failing field (field is missing
// when the body itself is wrong).
export function validateRoom(body) {
  if (body === null || typeof body !== 'object' || Array.isArray(body)) {
    return { error: 'Body must be a JSON object' };
  }
  const name = typeof body.name === 'string' ? body.name.trim() : '';
  if (name === '') return { error: 'Name is required', field: 'name' };
  if (name.length > NAME_MAX) {
    return { error: `Name must be ${NAME_MAX} characters or fewer`, field: 'name' };
  }
  const room = { name };
  for (const { field, label, min, max } of SIZES) {
    const value = body[field];
    if (typeof value !== 'number' || !Number.isFinite(value) || value < min || value > max) {
      return { error: `${label} must be a number from ${min} to ${max} m`, field };
    }
    room[field] = value;
  }
  return { room };
}
```

Notes for the implementer:

| Point | Why |
|---|---|
| `name ... UNIQUE COLLATE NOCASE` | the column's collation makes both the uniqueness and `ORDER BY name` ignore case |
| `nameUsed` query checked before insert/update | gives a clean `NameTakenError` instead of parsing SQLite error codes; the database is synchronous so there is no race |
| `id IS NOT ?` | with `null` (create) it matches every row; with an id (update) it skips the room being edited |
| CHECK constraints | a second guard; the API validates first |
| `now` parameter | lets tests fix `createdAt`/`updatedAt` |

`server/db.js`:

```js
// The farm database: schema and room queries.

import { DatabaseSync } from 'node:sqlite';

const SCHEMA = `
CREATE TABLE IF NOT EXISTS rooms (
  id         INTEGER PRIMARY KEY,
  name       TEXT    NOT NULL UNIQUE COLLATE NOCASE,
  length_m   REAL    NOT NULL CHECK (length_m BETWEEN 0.5 AND 100),
  width_m    REAL    NOT NULL CHECK (width_m BETWEEN 0.5 AND 100),
  height_m   REAL    NOT NULL CHECK (height_m BETWEEN 0.5 AND 10),
  created_at INTEGER NOT NULL,
  updated_at INTEGER NOT NULL
) STRICT;
`;

// Thrown by createRoom and updateRoom when another room already has the name
// (ignoring case).
export class NameTakenError extends Error {
  constructor(name) {
    super(`A room called "${name}" already exists`);
    this.name = 'NameTakenError';
    this.roomName = name;
  }
}

function toRoom(row) {
  return {
    id: row.id,
    name: row.name,
    length: row.length_m,
    width: row.width_m,
    height: row.height_m,
    volume: Math.round(row.length_m * row.width_m * row.height_m * 100) / 100,
    createdAt: row.created_at,
    updatedAt: row.updated_at,
  };
}

// path is a file path or ':memory:'. Fields passed in must already be valid
// (see validateRoom in rooms.js).
export function openDb(path) {
  const db = new DatabaseSync(path);
  db.exec(SCHEMA);

  const q = {
    list: db.prepare('SELECT * FROM rooms ORDER BY name, id'),
    get: db.prepare('SELECT * FROM rooms WHERE id = ?'),
    nameUsed: db.prepare('SELECT 1 FROM rooms WHERE name = ? AND id IS NOT ?'),
    insert: db.prepare(`INSERT INTO rooms (name, length_m, width_m, height_m, created_at, updated_at)
                        VALUES (?, ?, ?, ?, ?, ?)`),
    update: db.prepare(`UPDATE rooms SET name = ?, length_m = ?, width_m = ?, height_m = ?, updated_at = ?
                        WHERE id = ?`),
    delete: db.prepare('DELETE FROM rooms WHERE id = ?'),
  };

  function checkName(name, id) {
    if (q.nameUsed.get(name, id)) throw new NameTakenError(name);
  }

  function getRoom(id) {
    const row = q.get.get(id);
    return row ? toRoom(row) : null;
  }

  return {
    listRooms() {
      return q.list.all().map(toRoom);
    },

    getRoom,

    createRoom({ name, length, width, height }, now = Date.now()) {
      checkName(name, null);
      const { lastInsertRowid } = q.insert.run(name, length, width, height, now, now);
      return getRoom(Number(lastInsertRowid));
    },

    // Returns the updated room, or null when there is no room with that id.
    updateRoom(id, { name, length, width, height }, now = Date.now()) {
      if (!q.get.get(id)) return null;
      checkName(name, id);
      q.update.run(name, length, width, height, now, id);
      return getRoom(id);
    },

    // Returns true when a room was deleted.
    deleteRoom(id) {
      return q.delete.run(id).changes > 0;
    },

    close() {
      db.close();
    },
  };
}
```

```js
// JSON API: /api/rooms and /api/rooms/:id.

import { validateRoom } from './rooms.js';
import { NameTakenError } from './db.js';

const MAX_BODY = 64 * 1024;

class HttpError extends Error {
  constructor(status, message, field) {
    super(message);
    this.status = status;
    this.field = field;
  }
}

function send(res, status, data) {
  res.writeHead(status, { 'content-type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

async function readJson(req) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > MAX_BODY) throw new HttpError(413, 'Body is too large');
    chunks.push(chunk);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } catch {
    throw new HttpError(400, 'Body must be valid JSON');
  }
}

async function readRoom(req) {
  const result = validateRoom(await readJson(req));
  if (result.error) throw new HttpError(400, result.error, result.field);
  return result.room;
}

const notFound = () => new HttpError(404, 'Not found');
const noRoom = () => new HttpError(404, 'Room not found');

// Returns a handler for every request whose path starts with /api.
export function createApi(db) {
  async function route(req, res, path) {
    if (path === '/api/rooms') {
      if (req.method === 'GET') return send(res, 200, db.listRooms());
      if (req.method === 'POST') return send(res, 201, db.createRoom(await readRoom(req)));
      throw notFound();
    }
    const match = /^\/api\/rooms\/([1-9][0-9]{0,14})$/.exec(path);
    if (!match) throw notFound();
    const id = Number(match[1]);
    if (req.method === 'PUT') {
      if (!db.getRoom(id)) throw noRoom();
      return send(res, 200, db.updateRoom(id, await readRoom(req)));
    }
    if (req.method === 'DELETE') {
      if (!db.deleteRoom(id)) throw noRoom();
      res.writeHead(204);
      return res.end();
    }
    throw notFound();
  }

  return async function api(req, res, url) {
    try {
      await route(req, res, url.pathname);
    } catch (err) {
      if (err instanceof HttpError) {
        send(res, err.status, err.field ? { error: err.message, field: err.field } : { error: err.message });
      } else if (err instanceof NameTakenError) {
        send(res, 409, { error: err.message, field: 'name' });
      } else {
        console.error(err);
        send(res, 500, { error: 'Something went wrong' });
      }
    }
  };
}
```

import { readFile } from 'node:fs/promises';
import { join } from 'node:path';

const TYPES = {
  '.html': 'text/html; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
};

function notFound(res) {
  res.writeHead(404, { 'content-type': 'text/plain; charset=utf-8' });
  res.end('Not found');
}

export function createStatic(dir) {
  return async function serve(req, res, url) {
    const name = url.pathname === '/' ? 'index.html' : url.pathname.slice(1);
    const ext = /^[a-z0-9-]+(\.[a-z]+)$/.exec(name)?.[1];
    if ((req.method !== 'GET' && req.method !== 'HEAD') || !TYPES[ext]) return notFound(res);
    let data;
    try {
      data = await readFile(join(dir, name));
    } catch (err) {
      if (err.code === 'ENOENT') return notFound(res);
      throw err;
    }
    res.writeHead(200, { 'content-type': TYPES[ext], 'cache-control': 'no-cache' });
    res.end(req.method === 'HEAD' ? undefined : data);
  };
}

function cell(text, label, className) {
  const td = document.createElement('td');
  td.textContent = text;
  td.dataset.label = label;
  if (className) td.className = className;
  return td;
}

function button(text, action, room, className) {
  const b = document.createElement('button');
  b.type = 'button';
  b.textContent = text;
  b.dataset.action = action;
  b.dataset.id = room.id;
  b.setAttribute('aria-label', `${text} ${room.name}`);
  if (className) b.className = className;
  return b;
}

function row(room) {
  const tr = document.createElement('tr');
  const actions = document.createElement('td');
  actions.className = 'row-actions';
  actions.append(button('Edit', 'edit', room), button('Delete', 'delete', room, 'danger-text'));
  tr.append(
    cell(room.name, 'Name', 'name'),
    cell(`${metres.format(room.length)} × ${metres.format(room.width)} × ${metres.format(room.height)}`, 'Size (m)'),
    cell(cubic.format(room.volume), 'Volume (m³)', 'num'),
    cell(day.format(room.createdAt), 'Created'),
    actions,
  );
  return tr;
}

function render() {
  els.table.tBodies[0].replaceChildren(...rooms.map(row));
  els.table.hidden = rooms.length === 0;
  els.empty.hidden = !loaded || rooms.length > 0;
}

// ---- add / edit dialog ----

function clearErrors() {
  els.formError.hidden = true;
  for (const f of FIELDS) {
    errors[f].textContent = '';
    inputs[f].removeAttribute('aria-invalid');
  }
}

function updateVolume() {
  const v = inputs.length.valueAsNumber * inputs.width.valueAsNumber * inputs.height.valueAsNumber;
  els.volume.textContent = Number.isFinite(v) ? `Volume: ${cubic.format(v)} m³` : 'Volume: — m³';
}

function openRoomDialog(room) {
  hideBanner();
  editingId = room ? room.id : null;
  els.title.textContent = room ? 'Edit room' : 'Add room';
  els.form.reset();
  if (room) for (const f of FIELDS) inputs[f].value = room[f];
  clearErrors();
  updateVolume();
  els.dialog.showModal();
}

function showFormError(err) {
  if (err.field && inputs[err.field]) {
    errors[err.field].textContent = err.message;
    inputs[err.field].setAttribute('aria-invalid', 'true');
    inputs[err.field].focus();
  } else {
    els.formError.textContent = err.message;
    els.formError.hidden = false;
  }
}

async function roomGone() {
  els.dialog.close();
  showBanner(GONE);
  await load();
}

els.form.addEventListener('submit', async (event) => {
  event.preventDefault();
  clearErrors();
  const body = {
    name: inputs.name.value,
    length: inputs.length.valueAsNumber,
    width: inputs.width.valueAsNumber,
    height: inputs.height.valueAsNumber,
  };
  els.save.disabled = true;
  try {
    if (editingId === null) await request('POST', '/api/rooms', body);
    else await request('PUT', `/api/rooms/${editingId}`, body);
    els.dialog.close();
    await load();
  } catch (err) {
    if (err.status === 404) await roomGone();
    else if (err.status === 400 || err.status === 409) showFormError(err);
    else showFormError(new ApiError(0, { error: `${OFFLINE}. Try again.` }));
  } finally {
    els.save.disabled = false;
  }
});

els.form.addEventListener('input', updateVolume);
els.cancel.addEventListener('click', () => els.dialog.close());

// ---- delete ----

function openDeleteDialog(room) {
  hideBanner();
  deletingId = room.id;
  els.deleteText.textContent = `Delete ${room.name}? This can't be undone.`;
  els.deleteDialog.returnValue = '';
  els.deleteDialog.showModal();
}

els.deleteDialog.addEventListener('close', async () => {
  if (els.deleteDialog.returnValue !== 'delete') return;
  try {
    await request('DELETE', `/api/rooms/${deletingId}`);
    await load();
  } catch (err) {
    if (err.status === 404) {
      showBanner(GONE);
      await load();
    } else {
      showBanner(OFFLINE, { retry: true });
    }
  }
});

// ---- wiring ----

els.add.addEventListener('click', () => openRoomDialog(null));
els.addEmpty.addEventListener('click', () => openRoomDialog(null));
els.retry.addEventListener('click', load);

els.table.addEventListener('click', (event) => {
  const b = event.target.closest('button[data-action]');
  if (!b) return;
  const room = rooms.find((r) => r.id === Number(b.dataset.id));
  if (!room) return;
  if (b.dataset.action === 'edit') openRoomDialog(room);
  else openDeleteDialog(room);
});

load();
