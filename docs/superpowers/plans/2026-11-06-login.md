# Login Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Everyone signs in to the farm app (a worker with their name and a 6-digit PIN, an owner with a password); a worker does floor work and never sees money; the person signed in is the one recorded; then the farm app can go on Tailscale Funnel.

**Architecture:** Spec `docs/superpowers/specs/2026-11-06-login-design.md`. Database 25 adds the sign-in columns to `workers` and a `sessions` table. `server/auth.js` hashes secrets (scrypt), checks PINs and passwords, locks accounts and addresses, and finds the session of a request's cookie. `server/app.js` answers the sign-in routes (`server/auth-api.js`) and then gates every other `/api` request: no session 401, a worker on a route not in `server/roles.js`'s allow-list 403, and a worker's JSON replies pass through `roles.js`'s filter, which drops every money key. `server/accounts-api.js` gives the owner the Settings → Workers controls; `server/owner.js` sets the first owner from the VM's shell. In the browser, `web/boot.js` asks `GET /api/session` first and either shows the sign-in view (`web/login.js`) or loads the app with the role on `<html data-role>`.

**Tech Stack:** Node 22.13+, `node:http`, `node:sqlite`, `node:crypto` (scrypt, SHA-256, random bytes), plain HTML/CSS/JS modules, no build, no npm dependencies, `node --test`.

    account(id) {

    // those not expired at now.
    accounts(now) {
      const counts = new Map(q.sessionCounts.all(now).map((r) => [r.worker_id, r.n]));
      return q.workers.all().map((r) => ({
        id: r.id, name: r.name, active: r.active === 1, role: r.role, hasSecret: r.secret_hash !== null, lockedUntil: r.locked_until,
        sessions: counts.get(r.id) ?? 0,

// Signing in's parts (spec 2026-11-06-login-design.md 3): the scrypt hash, the PIN and password rules, the token, the cookie

import { test } from 'node:test';
import assert from 'node:assert/strict';
import {
  hashSecret, checkSecret, pinError, passwordError, newToken, tokenHash, readCookie, sessionCookie, clearCookie, clientAddress, COOKIE,
} from '../server/auth.js';

test('a hashed secret checks against itself only; each hash has its own salt', async () => {
  const h = await hashSecret('482913');
  assert.match(h, /^scrypt\$32768\$8\$1\$[A-Za-z0-9+/=]+\$[A-Za-z0-9+/=]+$/);
  assert.equal(await checkSecret('482913', h), true);
  assert.equal(await checkSecret('482914', h), false);
  assert.notEqual(await hashSecret('482913'), h);
  assert.equal(await checkSecret('482913', null), false);
  assert.equal(await checkSecret('482913', 'garbage'), false);
  assert.equal(await checkSecret(undefined, h), false);
});

test('a PIN is 6 digits, not one digit six times, not 123456 or 654321', () => {
  assert.equal(pinError('482913'), null);
  for (const bad of ['48291', '4829134', '48291a', '', 482913]) assert.equal(pinError(bad), 'The PIN must be 6 digits', String(bad));
  for (const easy of ['000000', '777777', '123456', '654321']) assert.equal(pinError(easy), 'Pick a PIN that is harder to guess', easy);
});

test('a password is 12 to 200 characters', () => {
  assert.equal(passwordError('a'.repeat(12)), null);
  assert.equal(passwordError('a'.repeat(11)), 'The password must be at least 12 characters');
  assert.equal(passwordError('a'.repeat(201)), 'The password must be at most 200 characters');
  assert.equal(passwordError(null), 'The password must be at least 12 characters');
});

test('a token is 32 random bytes in base64url; only its SHA-256 is kept', () => {
  const t = newToken();
  assert.match(t, /^[A-Za-z0-9_-]{43}$/);
  assert.notEqual(newToken(), t);
  assert.match(tokenHash(t), /^[0-9a-f]{64}$/);
  assert.equal(tokenHash(t), tokenHash(t));
});

test('the cookie: read from among others; set HttpOnly, Secure, SameSite=Strict for its length; cleared', () => {
  assert.equal(readCookie(`theme=dark; ${COOKIE}=abc_-1; x=y`), 'abc_-1');

// token and cookie, the client's address; createAuth (below) signs in and out and finds the session of a request.

import { scrypt, randomBytes, timingSafeEqual, createHash } from 'node:crypto';
import { promisify } from 'node:util';

const scryptAsync = promisify(scrypt);
const COST = { N: 2 ** 15, r: 8, p: 1 };
const KEY_BYTES = 32;
const MAXMEM = 64 * 1024 * 1024; // N 2^15 with r 8 needs 32 MiB, Node's default limit exactly
export const COOKIE = 'farm_session';
const DAY_MS = 86_400_000;
export const SESSION_MS = { worker: 30 * DAY_MS, owner: 7 * DAY_MS };

// → 'scrypt$N$r$p$<salt>$<hash>' (base64).
export async function hashSecret(secret) {
  const salt = randomBytes(16);
  const hash = await scryptAsync(secret.normalize('NFC'), salt, KEY_BYTES, { ...COST, maxmem: MAXMEM });
  return `scrypt$${COST.N}$${COST.r}$${COST.p}$${salt.toString('base64')}$${hash.toString('base64')}`;
}

// → true when secret is the one stored hashed (a stored value not of hashSecret's shape: false).
export async function checkSecret(secret, stored) {
  const parts = typeof stored === 'string' ? stored.split('$') : [];
  if (parts.length !== 6 || parts[0] !== 'scrypt' || typeof secret !== 'string') return false;
  const [N, r, p] = parts.slice(1, 4).map(Number);
  const salt = Buffer.from(parts[4], 'base64');
  const want = Buffer.from(parts[5], 'base64');
  if (!(N > 1 && r > 0 && p > 0) || want.length === 0) return false;
  const got = await scryptAsync(secret.normalize('NFC'), salt, want.length, { N, r, p, maxmem: MAXMEM });
  return timingSafeEqual(got, want);
}

// A worker's PIN: exactly 6 digits, not one digit six times, not 123456 or 654321. → the error, or null.
export function pinError(pin) {
  if (typeof pin !== 'string' || !/^[0-9]{6}$/.test(pin)) return 'The PIN must be 6 digits';
  if (/^(\d)\1{5}$/.test(pin) || pin === '123456' || pin === '654321') return 'Pick a PIN that is harder to guess';
  return null;
}

// An owner's password: 12 to 200 characters. → the error, or null.
export function passwordError(password) {
  const length = typeof password === 'string' ? [...password].length : 0;
  if (length < 12) return 'The password must be at least 12 characters';
  if (length > 200) return 'The password must be at most 200 characters';
  return null;
}

export const newToken = () => randomBytes(32).toString('base64url');
export const tokenHash = (token) => createHash('sha256').update(token).digest('hex');

// The value of the cookie name in a Cookie header, or null.
export function readCookie(header, name = COOKIE) {
  for (const part of String(header ?? '').split(';')) {
    const at = part.indexOf('=');
    if (at > 0 && part.slice(0, at).trim() === name) return part.slice(at + 1).trim() || null;
  }
  return null;
}

export const sessionCookie = (token, ms) => `${COOKIE}=${token}; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age=${Math.floor(ms / 1000)}`;
export const clearCookie = () => `${COOKIE}=; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age=0`;

// Who is asking, for the per-address limit: behind Tailscale (serve or Funnel) the request comes from this machine and
// X-Forwarded-For names the client; from anywhere else the header is the client's own word and is ignored.
export function clientAddress(req) {
  const socket = req.socket?.remoteAddress ?? '';
  const local = socket === '127.0.0.1' || socket === '::1' || socket === '::ffff:127.0.0.1';
  const forwarded = local ? String(req.headers['x-forwarded-for'] ?? '').split(',')[0].trim() : '';
  return forwarded || socket;
}

export const LOCK = { tries: 5, ms: 15 * 60_000 };
export const ADDRESS_LIMIT = { tries: 20, ms: 15 * 60_000 };
const TOUCH_MS = 3_600_000; // a session's last_seen and expiry are written at most once an hour

// db: the farm database. log(line): one line per sign-in try (never the secret). → { names(), login(req, body), who(req,
// res), logout(req), changePassword(req, me, body), prune(), close() }. login and changePassword → { status, body, cookie? }.
export function createAuth({ db, now = Date.now, log = (line) => console.log(line) }) {
  const tries = new Map(); // address → { count, until }: the tries of the address's current 15 minutes
  // The hash an unknown name is checked against, so it takes as long as a known one; made at start, so the first unknown
  // name after a restart is not slower.
  const dummy = hashSecret(newToken());

  const wrong = (role) => ({ status: 401, body: { error: role === 'owner' ? 'That password is not right' : 'That PIN is not right', field: 'secret' } });
  const lockedReply = (until) => ({ status: 429, body: { error: `Too many tries. Try again at ${localTime(until)}` } });

  // Counts the try; → true when the address has had more than its tries in its window.
  function overLimit(address, t) {
    const seen = tries.get(address);
    if (!seen || seen.until <= t) {
      tries.set(address, { count: 1, until: t + ADDRESS_LIMIT.ms });
      return false;
    }
    seen.count += 1;
    return seen.count > ADDRESS_LIMIT.tries;
  }

  function startSession(req, account, t) {
    const token = newToken();
    const ms = SESSION_MS[account.role];
    const userAgent = String(req.headers['user-agent'] ?? '').slice(0, 200) || null;
    db.addSession({ tokenHash: tokenHash(token), workerId: account.id, at: t, expiresAt: t + ms, userAgent });
    return sessionCookie(token, ms);
  }

  async function login(req, body) {
    const t = now();
    const address = clientAddress(req);
    const id = Number.isSafeInteger(body?.id) ? body.id : null;
    const secret = typeof body?.secret === 'string' ? body.secret : '';
    const note = (result) => log(`login ${new Date(t).toISOString()} ${address} worker ${id ?? '-'} ${result}`);
    if (overLimit(address, t)) {
      note('address-limited');
      return { status: 429, body: { error: 'Too many tries from here. Try again in 15 minutes' } };
    }
    const account = id === null ? null : db.account(id);
    const usable = Boolean(account?.active && account.secretHash !== null);
    if (usable && account.lockedUntil !== null && account.lockedUntil > t) {
      note('locked');
      return lockedReply(account.lockedUntil);
    }
    const right = await checkSecret(secret, usable ? account.secretHash : await dummy);
    if (!usable || !right) {
      note('wrong');
      if (!usable) return wrong(account?.role);
      const count = account.failedTries + 1;
      if (count >= LOCK.tries) {
        db.setTries(account.id, 0, t + LOCK.ms);
        return lockedReply(t + LOCK.ms);
      }
      db.setTries(account.id, count, null);
      return wrong(account.role);
    }
    db.setTries(account.id, 0, null);
    note('right');
    return { status: 200, body: { id: account.id, name: account.name, role: account.role }, cookie: startSession(req, account, t) };
  }

  // The person of the request's session: { id, name, role, sessionId }, or null. An hour or more since it was last seen,
  // its expiry moves on and res gets the cookie again.
  function who(req, res) {
    const token = readCookie(req.headers.cookie);
    if (!token) return null;
    const t = now();
    const s = db.session(tokenHash(token));
    if (!s) return null;
    if (!s.active || s.expiresAt <= t) {
      db.endSession(s.id);
      return null;
    }
    if (t - s.lastSeen >= TOUCH_MS) {
      const ms = SESSION_MS[s.role];
      db.touchSession(s.id, t, t + ms);
      res.setHeader('set-cookie', sessionCookie(token, ms));
    }
    return { id: s.workerId, name: s.name, role: s.role, sessionId: s.id };
  }

  function logout(req) {
    const token = readCookie(req.headers.cookie);
    const s = token && db.session(tokenHash(token));
    if (s) db.endSession(s.id);
  }

  // An owner's own password { current, next }: every session of theirs ends and this browser (req) gets a new one.
  async function changePassword(req, me, body) {
    const account = db.account(me.id);
    const current = typeof body?.current === 'string' ? body.current : '';
    if (!account || !(await checkSecret(current, account.secretHash))) {
      return { status: 400, body: { error: 'That password is not right', field: 'current' } };
    }
    const problem = passwordError(body?.next);
    if (problem) return { status: 400, body: { error: problem, field: 'next' } };
    db.setSecret(account.id, await hashSecret(body.next));
    return { status: 204, body: null, cookie: startSession(req, account, now()) };
  }

  function prune() {
    const t = now();
    db.pruneSessions(t);
    for (const [address, seen] of tries) if (seen.until <= t) tries.delete(address);
  }

  prune();
  const timer = setInterval(prune, TOUCH_MS);
  timer.unref();
  return { names: () => db.loginNames(), login, who, logout, changePassword, prune, close: () => clearInterval(timer) };
}

export const MONEY_KEYS = new Set([
  'value', 'cost', 'costPerKg', 'waterCost', 'prices', 'electricityPrice', 'waterPrice', 'priceKg', 'rmMonth', 'unitCost', 'price',
  'total', 'blockCost', 'pricePerBlock', 'profit', 'income', 'unpriced', 'missing',
]);

// The settings a worker's pages read: how often to read again. (No prices, no Telegram chat or token, no gateway.)
const WORKER_SETTINGS = ['pollSeconds'];

function withoutMoney(v) {
  if (Array.isArray(v)) return v.map(withoutMoney);
  if (v !== null && typeof v === 'object') {
    return Object.fromEntries(Object.entries(v).filter(([k]) => !MONEY_KEYS.has(k)).map(([k, x]) => [k, withoutMoney(x)]));
  }
  return v;
}

// A worker's reply of path: the settings cut to WORKER_SETTINGS, every reply without its money.
export function workerReply(path, data) {
  if (path === '/api/settings' && data !== null && typeof data === 'object') {
    return Object.fromEntries(WORKER_SETTINGS.filter((k) => k in data).map((k) => [k, data[k]]));
  }
  return withoutMoney(data);
}

// Makes res pass its JSON replies (a 2xx with an application/json type) through workerReply on their way out; a CSV, an
// event stream, an error and an empty reply go as they are. The API's send() gives no Content-Length, so the length may change.
export function filterWorkerReplies(res, path) {
  let json = false;
  const writeHead = res.writeHead.bind(res);
  res.writeHead = (status, ...rest) => {
    const headers = rest.find((x) => x !== null && typeof x === 'object') ?? {};
    const type = headers['content-type'] ?? res.getHeader('content-type') ?? '';
    json = status < 300 && /^application\/json\b/i.test(String(type));
    return writeHead(status, ...rest);
  };
  const end = res.end.bind(res);
  res.end = (chunk, ...rest) => {
    if (json && chunk !== undefined && chunk !== null) chunk = JSON.stringify(workerReply(path, JSON.parse(String(chunk))));
    return end(chunk, ...rest);
  };
}

// Settings → Workers' sign-in controls (spec 2026-11-06-login-design.md 5), the owner's only (roles.js leaves them out of a
// worker's routes): GET /api/workers/accounts, POST /api/workers/:id/secret { secret }, PUT /api/workers/:id/role { role },
// POST /api/workers/:id/unlock and POST /api/workers/:id/sign-out. Each change answers { account } as the list has it.

import { hashSecret, pinError, passwordError } from './auth.js';

const ONE = /^\/api\/workers\/([1-9][0-9]{0,14})\/(secret|role|unlock|sign-out)$/;

// → async (req, res, path, { send, readJson, HttpError }) that answers true when the path is one of its routes.
export function createAccountRoutes({ db, now = Date.now }) {
  const accountOf = (id) => db.accounts(now()).find((a) => a.id === id);

  return async function accountRoutes(req, res, path, { send, readJson, HttpError }) {
    if (path === '/api/workers/accounts') {
      if (req.method !== 'GET') throw new HttpError(404, 'Not found');
      send(res, 200, { accounts: db.accounts(now()) });
      return true;
    }
    const match = ONE.exec(path);
    if (!match) return false;
    const [, idText, what] = match;
    if (req.method !== (what === 'role' ? 'PUT' : 'POST')) throw new HttpError(404, 'Not found');
    const account = db.account(Number(idText));
    if (!account) throw new HttpError(404, 'Worker not found');
    if (what === 'secret') {
      const secret = (await readJson(req))?.secret;
      const problem = account.role === 'owner' ? passwordError(secret) : pinError(secret);
      if (problem) throw new HttpError(400, problem, 'secret');
      db.setSecret(account.id, await hashSecret(secret));
    } else if (what === 'role') {
      const role = (await readJson(req))?.role;
      if (role !== 'owner' && role !== 'worker') throw new HttpError(400, 'The role must be owner or worker', 'role');
      if (role === 'worker' && account.role === 'owner' && account.active && db.ownerCount() <= 1) {
        throw new HttpError(400, 'The farm needs an owner', 'role');
      }
      // A worker's PIN is no owner's password, nor the other way: a new role needs a new secret (Set PIN or Set password).
      if (role !== account.role) {
        db.setRole(account.id, role);
        db.setSecret(account.id, null);
      }
    } else if (what === 'unlock') {
      db.setTries(account.id, 0, null);
    } else {
      db.signOutEverywhere(account.id);
    }
    send(res, 200, { account: accountOf(account.id) });
    return true;
  };
}

// → { id } or { error }.
export async function setOwnerPassword(db, name, password) {
  const trimmed = typeof name === 'string' ? name.trim() : '';
  if (!trimmed || trimmed.length > MAX_NAME || /[\u0000-\u001f\u007f]/.test(trimmed)) return { error: `Give a name of 1 to ${MAX_NAME} characters` };

// One line typed at the terminal, not shown.
function askHidden(question) {
  return new Promise((resolve, reject) => {
    const { stdin, stdout } = process;
    stdout.write(question);
    stdin.setRawMode(true);
    stdin.setEncoding('utf8');
    stdin.resume();
    let text = '';
    const done = () => { stdin.off('data', onData); stdin.setRawMode(false); stdin.pause(); stdout.write('\n'); };
    function onData(chunk) {
      for (const ch of chunk) {
        if (ch === '\r' || ch === '\n') { done(); resolve(text); return; }
        if (ch === '\u0003') { done(); reject(new Error('Stopped')); return; } // Ctrl+C
        text = ch === '\u007f' || ch === '\b' ? text.slice(0, -1) : text + ch;
      }
    }
    stdin.on('data', onData);
  });
}

  if (!process.stdin.isTTY) {
    const [first = '', second = ''] = readFileSync(0, 'utf8').split(/\r?\n/);
    return [first, second];
  }
  return [await askHidden('Password (12 characters or more): '), await askHidden('Again: ')];
}

    <div class="login-actions">

      <button type="submit" class="primary" data-submit>Sign in</button>
    </div>
  </form>
</section>`;

export function mount(container, { onSignedIn }) {
  container.innerHTML = TEMPLATE;
  const q = (s) => container.querySelector(s);
  const els = {
    lead: q('[data-lead]'), names: q('[data-names]'), loadError: q('[data-load-error]'), retry: q('[data-retry]'), form: q('[data-form]'),
    label: q('[data-label]'), input: q('#login-secret'), hint: q('[data-hint]'), error: q('[data-error]'), back: q('[data-back]'),
    submit: q('[data-submit]'),
  };
  let alive = true;
  let picked = null;
  let busy = false;
  document.title = 'Sign in · Mushroom farm';

  function nameButton(n) {
    const li = document.createElement('li');
    const b = document.createElement('button');
    b.type = 'button';
    b.className = 'login-name';
    b.textContent = n.name;
    b.addEventListener('click', () => pick(n));
    li.append(b);
    return li;
  }

  async function loadNames() {
    els.loadError.textContent = '';
    els.retry.hidden = true;
    try {
      const { names } = await request('GET', '/api/login/names');
      if (!alive) return;
      els.names.replaceChildren(...names.map(nameButton));
      els.lead.textContent = names.length ? 'Who are you?' : 'Nobody can sign in yet: the owner sets the first password on the farm computer (npm run owner).';
      els.names.querySelector('button')?.focus();
    } catch (err) {
      if (!alive) return;
      els.loadError.textContent = failText(err.status, err.message);
      els.retry.hidden = false;
    }
  }

  function showError(text) {
    els.error.textContent = text;
    els.input.setAttribute('aria-invalid', 'true');
    els.input.focus();
  }

  function pick(n) {
    picked = n;
    const box = SECRET[n.kind];
    els.lead.textContent = `Signing in as ${n.name}`;
    els.names.hidden = true;
    els.form.hidden = false;
    els.label.textContent = box.label;
    els.input.value = '';
    els.input.setAttribute('inputmode', box.inputmode);
    els.input.setAttribute('autocomplete', box.autocomplete);
    els.input.maxLength = box.maxlength;
    els.hint.textContent = box.hint ?? '';
    els.hint.hidden = !box.hint;
    els.error.textContent = '';
    els.input.removeAttribute('aria-invalid');
    els.input.focus();
  }

  els.back.addEventListener('click', () => {
    picked = null;
    els.form.hidden = true;
    els.names.hidden = false;
    els.lead.textContent = 'Who are you?';
    els.names.querySelector('button')?.focus();
  });
  els.retry.addEventListener('click', loadNames);
  els.form.addEventListener('submit', async (e) => {
    e.preventDefault();
    if (busy || !picked) return;
    const secret = els.input.value;
    const problem = secretProblem(picked.kind, secret);
    if (problem) return showError(problem);
    busy = true;
    els.submit.disabled = true;

      if (alive) onSignedIn(who);

      if (alive) {

        showError(failText(err.status, err.message));
      }
    } finally {
      busy = false;
      if (alive) els.submit.disabled = false;
    }
  });

  loadNames();
  return function unmount() {
    alive = false;
    container.replaceChildren();
  };
}

// "Worker · signed in on 2 devices", "Owner · no password yet: cannot sign in", "Worker · locked until 10:10".
export function accountLine(a, now) {
